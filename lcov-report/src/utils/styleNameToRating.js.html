/**
 * Returns the rating for a given style name (style.name / styleName).
 * @param {string} styleName - Raw style name from API (e.g. "miniaturJarv1", "pixar-style")
 * @returns {number | undefined} Rating if in map, otherwise undefined
 */
export function getRatingForStyleName(styleName) {
  if (styleName == null || styleName === '') return undefined;
  return STYLE_NAME_RATING_MAP[styleName];
}

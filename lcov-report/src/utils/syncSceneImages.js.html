/**
 * Fast and optimized scene-timeline image synchronization
 * Ensures timeline elements have correct images from scenes
 * @param {Object} storyData - The story data
 * @param {Object} store - The MobX store
 * @param {Array} scenes - The scenes array
 * @param {boolean} skipFinalSave - If true, skip the final save call (caller will handle it)
 */
 
export const syncSceneImages = async (storyData, store, scenes, skipFinalSave = false) => {
  if (!storyData || !store || !scenes || scenes.length === 0) {
    return;
  }
 
  const syncStart = performance.now();
 
  // Get current timeline elements (both images and videos)
  const timelineElements = store.editorElements || [];
  const visualElements = timelineElements.filter(
    (el) => el.type === 'imageUrl' || el.type === 'video',
  );
 
  // Build set of valid scene IDs for quick lookup
  const validSceneIds = new Set(scenes.map((scene) => scene._id));
 
  // Build scene lookup map for O(1) access
  const sceneMap = new Map();
  scenes.forEach((scene) => {
    if (scene._id && scene.selectedImage?.url) {
      // Check if this is a video selection (has videoData)
      const isVideo = scene.selectedImage.videoData?.type === 'video';
      // Prefer CDN URL (googleCloudUrl) — this is what store.updateCanvasImage stores as properties.src
      const resolvedUrl = scene.selectedImage.googleCloudUrl || scene.selectedImage.url;
      const resolvedMinUrl = scene.selectedImage.minGoogleCloudUrl || scene.selectedImage.minUrl;
 
      sceneMap.set(scene._id, {
        url: resolvedUrl,
        minUrl: resolvedMinUrl,
        imageId: scene.selectedImage.id,
        prompt: scene.selectedImage.prompt || '',
        negativePrompt: scene.selectedImage.negativePrompt || '',
        imageHeight: scene.selectedImage.imageHeight || 0,
        imageWidth: scene.selectedImage.imageWidth || 0,
        isVideo,
        videoData: scene.selectedImage.videoData || null,
      });
    }
  });
 
  // Find orphaned elements (elements whose scenes no longer exist)
  const orphanedElements = visualElements.filter((element) => {
    // Only check row 0 imageUrl elements (main scene elements)
    if (element.type !== 'imageUrl' || element.row !== 0) {
      return false;
    }
    // Element is orphaned if its pointId doesn't match any existing scene
    return element.pointId && !validSceneIds.has(element.pointId);
  });
 
  // Remove orphaned elements
  if (orphanedElements.length > 0) {
    console.log(
      `[syncSceneImages] Removing ${orphanedElements.length} orphaned elements:`,
      orphanedElements.map((el) => ({ id: el.id, pointId: el.pointId })),
    );
 
    const orphanedIds = new Set(orphanedElements.map((el) => el.id));
    store.editorElements = store.editorElements.filter((el) => !orphanedIds.has(el.id));
  }
 
  // Find elements that need updating
  const elementsToUpdate = [];
  const elementsToAdd = [];
 
  visualElements.forEach((element) => {
    const sceneData = sceneMap.get(element.pointId);
 
    if (sceneData) {
      // Skip video elements - they should not be overwritten by scene sync
      // Video elements are explicitly set and should be preserved
      if (element.type === 'video') {
        return;
      }
 
      // Skip if scene has video data - don't replace video with image
      if (sceneData.isVideo) {
        return;
      }
 
      // Check if element needs updating (only for images)
      // Compare against resolved CDN URL (same as what properties.src stores)
      const needsUpdate =
        element.properties?.src !== sceneData.url ||
        element.properties?.minUrl !== sceneData.minUrl ||
        element.properties?.imageId !== sceneData.imageId;
 
      if (needsUpdate) {
        elementsToUpdate.push({
          element,
          sceneData,
          reason: `src: ${element.properties?.src?.substring(0, 30)}... -> ${sceneData.url?.substring(0, 30)}...`,
        });
      }
    }
  });
 
  // Find scenes that don't have timeline elements yet
  scenes.forEach((scene) => {
    if (scene._id && scene.selectedImage?.url) {
      // Check for both imageUrl AND video elements
      const hasTimelineElement = visualElements.some((el) => el.pointId === scene._id);
 
      // Skip if scene has video data - video elements are added separately
      const isVideoScene = scene.selectedImage.videoData?.type === 'video';
 
      if (!hasTimelineElement && !isVideoScene) {
        elementsToAdd.push({
          scene,
          reason: 'Missing timeline element for scene with image',
        });
      }
    }
  });
 
  if (elementsToUpdate.length === 0 && elementsToAdd.length === 0) {
    return;
  }
 
  // Parallel batch update existing elements
  if (elementsToUpdate.length > 0) {
    const updateStart = performance.now();
 
    const updatePromises = elementsToUpdate.map(async ({ element, sceneData, reason }) => {
      try {
        await store.updateCanvasImage({
          url: sceneData.url,
          minUrl: sceneData.minUrl,
          pointId: element.pointId,
          imageId: sceneData.imageId,
          skipAnimation: true, // Skip animations for sync operations
          skipSave: true, // Skip individual saves - we'll save at the end
        });
        return { success: true, elementId: element.id };
      } catch (error) {
        console.error(`[⏱️ syncSceneImages] Failed to update element ${element.id}:`, error);
        return { success: false, elementId: element.id, error };
      }
    });
 
    const updateResults = await Promise.allSettled(updatePromises);
    const successfulUpdates = updateResults.filter(
      (result) => result.status === 'fulfilled' && result.value.success,
    ).length;
  }
 
  // Parallel add missing elements (if any)
  if (elementsToAdd.length > 0) {
    const addStart = performance.now();
 
    const addPromises = elementsToAdd.map(async ({ scene, reason }) => {
      try {
        // Find appropriate timeframe for the new element
        const sceneIndex = scenes.findIndex((s) => s._id === scene._id);
        const timeFrame = {
          start: sceneIndex * 3000, // 3 seconds per scene as default
          end: (sceneIndex + 1) * 3000,
        };
 
        await store.addImageToCanvas({
          url: scene.selectedImage.url,
          minUrl: scene.selectedImage.minUrl,
          pointId: scene._id,
          imageId: scene.selectedImage.id,
          timeFrame,
          skipAnimation: true,
          skipSave: true,
        });
        return { success: true, sceneId: scene._id };
      } catch (error) {
        console.error(`[⏱️ syncSceneImages] Failed to add element for scene ${scene._id}:`, error);
        return { success: false, sceneId: scene._id, error };
      }
    });
 
    const addResults = await Promise.allSettled(addPromises);
    const successfulAdds = addResults.filter(
      (result) => result.status === 'fulfilled' && result.value.success,
    ).length;
  }
 
  // Single save at the end for better performance (skip if caller will handle it)
  if (elementsToUpdate.length > 0 || elementsToAdd.length > 0) {
    if (window.dispatchSaveTimelineState && !skipFinalSave) {
    }
  }
 
  const totalTime = performance.now() - syncStart;
};
 
/**
 * Check if scene-timeline sync is needed
 * Returns true if sync is required
 */
export const checkIfImageSyncNeeded = (storyData, store) => {
  if (!storyData?.scenes || !store?.editorElements) {
    return false;
  }
 
  const scenes = storyData.scenes;
  // Include both imageUrl and video elements
  const visualElements = store.editorElements.filter(
    (el) => el.type === 'imageUrl' || el.type === 'video',
  );
 
  // Build set of valid scene IDs
  const validSceneIds = new Set(scenes.map((scene) => scene._id));
 
  // Check for orphaned elements (elements whose scenes no longer exist)
  const hasOrphanedElements = visualElements.some(
    (el) => el.type === 'imageUrl' && el.row === 0 && el.pointId && !validSceneIds.has(el.pointId),
  );
 
  if (hasOrphanedElements) {
    return true;
  }
 
  // Quick check: if scene count doesn't match visual element count
  // Only count scenes that have images (not videos)
  const scenesWithImages = scenes.filter(
    (scene) => scene.selectedImage?.url && !scene.selectedImage?.videoData?.type,
  ).length;
  const imageElements = visualElements.filter((el) => el.type === 'imageUrl' && el.row === 0);
 
  if (scenesWithImages !== imageElements.length) {
    return true;
  }
 
  // Check if any scene has different image than timeline element
  for (const scene of scenes) {
    if (scene._id && scene.selectedImage?.url) {
      // Skip video scenes - they are handled separately
      if (scene.selectedImage?.videoData?.type === 'video') {
        continue;
      }
 
      const timelineElement = visualElements.find((el) => el.pointId === scene._id);
 
      if (!timelineElement) {
        return true;
      }
 
      // Only check src for imageUrl elements, not video elements
      // Compare against CDN URL (googleCloudUrl) — same as what properties.src stores
      const resolvedSceneUrl = scene.selectedImage.googleCloudUrl || scene.selectedImage.url;
      if (
        timelineElement.type === 'imageUrl' &&
        timelineElement.properties?.src !== resolvedSceneUrl
      ) {
        return true;
      }
    }
  }
 
  return false;
};

/**
 * Maps analysis segments (frozen at analyze time) onto the current script so TTS
 * receives edits made after analysis, especially ElevenLabs v3 emotion tags.
 *
 * The split shape must stay close to the original analyzer output: if the user
 * inserts `[emotion]` before or inside a segment, that tag should travel with
 * the segment instead of causing a fresh sentence split.
 *
 * @param {Array<{ text?: string, line?: string }>} segmentsInOrder
 * @param {string} _originalScript
 * @param {string} currentScript
 * @returns {string[]}
 */
export function resolveVoiceoverSegmentLinesForGeneration(
  segmentsInOrder,
  _originalScript,
  currentScript,
) {
  const curr = String(currentScript ?? '');
  if (!Array.isArray(segmentsInOrder) || segmentsInOrder.length === 0) {
    return [];
  }
 
  const lineFromSeg = (segment) => String(segment?.text ?? segment?.line ?? '');
  const trimLine = (line) => String(line ?? '').trim();
 
  if (segmentsInOrder.length === 1) {
    return [curr.trim() || lineFromSeg(segmentsInOrder[0])];
  }
 
  const splitBySentences = (value, targetCount) => {
    const parts = String(value ?? '')
      .match(/[^.!?]+[.!?]+(?:["')\]]+)?|[^.!?]+$/g)
      ?.map((part) => part.trim())
      .filter(Boolean);
 
    if (!parts || parts.length === 0) {
      return [];
    }
 
    if (parts.length <= targetCount) {
      return [...parts, ...Array(Math.max(0, targetCount - parts.length)).fill('')];
    }
 
    return Array.from({ length: targetCount }, (_, index) =>
      index < targetCount - 1 ? parts[index] || '' : parts.slice(index).join(' '),
    );
  };
 
  const stripEmotionTagsWithMap = (value) => {
    const text = String(value ?? '');
    let stripped = '';
    const map = [];
 
    for (let index = 0; index < text.length; index++) {
      if (text[index] === '[') {
        const closeIndex = text.indexOf(']', index + 1);
        if (closeIndex !== -1 && closeIndex - index <= 48) {
          index = closeIndex;
          continue;
        }
      }
 
      stripped += text[index];
      map.push(index);
    }
 
    const toOriginalStart = (strippedIndex) => {
      if (strippedIndex <= 0) return 0;
      if (strippedIndex >= map.length) return text.length;
      return map[strippedIndex];
    };
 
    const toOriginalEnd = (strippedIndex) => {
      if (strippedIndex <= 0) return 0;
      if (strippedIndex > map.length) return text.length;
      return map[strippedIndex - 1] + 1;
    };
 
    return { stripped, toOriginalStart, toOriginalEnd };
  };
 
  const currentWithoutEmotions = stripEmotionTagsWithMap(curr);
  const anchors = [];
  let strippedSearchFrom = 0;
 
  for (const segment of segmentsInOrder) {
    const base = trimLine(lineFromSeg(segment));
    if (!base) {
      anchors.push(null);
      continue;
    }
 
    const strippedBase = stripEmotionTagsWithMap(base).stripped.trim();
    if (!strippedBase) {
      anchors.push(null);
      continue;
    }
 
    let strippedStart = currentWithoutEmotions.stripped.indexOf(strippedBase, strippedSearchFrom);
    if (strippedStart === -1) {
      strippedStart = currentWithoutEmotions.stripped.indexOf(strippedBase);
    }
 
    if (strippedStart === -1) {
      anchors.push(null);
      continue;
    }
 
    const strippedEnd = strippedStart + strippedBase.length;
    anchors.push({
      start: currentWithoutEmotions.toOriginalStart(strippedStart),
      end: currentWithoutEmotions.toOriginalEnd(strippedEnd),
    });
    strippedSearchFrom = strippedEnd;
  }
 
  if (!anchors.some(Boolean)) {
    const sentenceSplit = splitBySentences(curr, segmentsInOrder.length);
    return sentenceSplit.length > 0
      ? sentenceSplit
      : segmentsInOrder.map((segment) => lineFromSeg(segment));
  }
 
  const result = Array(segmentsInOrder.length).fill('');
 
  const distributeGap = (gapText, fromIndex, toIndex) => {
    const count = toIndex - fromIndex + 1;
    if (count <= 0) return;
 
    const sentenceSplit = splitBySentences(gapText, count);
    if (sentenceSplit.length > 0) {
      for (let offset = 0; offset < count; offset++) {
        result[fromIndex + offset] = sentenceSplit[offset] || '';
      }
      return;
    }
 
    result[fromIndex] = trimLine(gapText);
  };
 
  let index = 0;
  while (index < segmentsInOrder.length) {
    const anchor = anchors[index];
 
    if (!anchor) {
      const runStart = index;
      while (index < segmentsInOrder.length && !anchors[index]) {
        index++;
      }
      const runEnd = index - 1;
      const previousAnchor = [...anchors.slice(0, runStart)].reverse().find(Boolean);
      const nextAnchor = anchors.slice(index).find(Boolean);
      const gapStart = previousAnchor?.end ?? 0;
      const gapEnd = nextAnchor?.start ?? curr.length;
      distributeGap(curr.slice(gapStart, gapEnd), runStart, runEnd);
      continue;
    }
 
    const previousIsAnchored = index > 0 && anchors[index - 1];
    const nextIsAnchored = index < anchors.length - 1 && anchors[index + 1];
    const hasUnmatchedBefore = index > 0 && !anchors[index - 1];
    const hasUnmatchedAfter = index < anchors.length - 1 && !anchors[index + 1];
 
    const start =
      index === 0
        ? 0
        : previousIsAnchored
          ? anchors[index - 1].end
          : hasUnmatchedBefore
            ? anchor.start
            : anchor.start;
    const end =
      index === anchors.length - 1
        ? curr.length
        : nextIsAnchored
          ? anchor.end
          : hasUnmatchedAfter
            ? anchor.end
            : anchor.end;
 
    result[index] = trimLine(curr.slice(start, end)) || lineFromSeg(segmentsInOrder[index]);
    index++;
  }
 
  return result;
}

// Simple audio processor for maintaining pitch while changing playback speed
class AudioProcessor {
  constructor() {
    this.audioContext = null;
    this.audioSources = new Map();
    this.isInitialized = false;
  }
 
  // Initialize Web Audio API context
  async initialize() {
    if (this.isInitialized) return;
 
    this.audioContext = new (window.AudioContext || window.webkitAudioContext)();
    
    // Resume context if suspended (required by modern browsers)
    if (this.audioContext.state === 'suspended') {
      await this.audioContext.resume();
    }
    
    this.isInitialized = true;
  }
 
  // Update playback rate for an audio element with pitch preservation
  async updateAudioPlaybackRate(audioElement, newRate) {
    if (!audioElement || !audioElement.src) return;
 
    try {
      // Initialize if needed
      if (!this.isInitialized) {
        await this.initialize();
      }
 
      const elementId = audioElement.id || audioElement.src;
      
      // Check if we already have a source for this element
      let audioData = this.audioSources.get(elementId);
      
      if (!audioData) {
        // Create new audio source only if it doesn't exist
        const source = this.audioContext.createMediaElementSource(audioElement);
        const gainNode = this.audioContext.createGain();
        
        // Connect the nodes
        source.connect(gainNode);
        gainNode.connect(this.audioContext.destination);
        
        // Store reference
        audioData = { source, gainNode, audioElement };
        this.audioSources.set(elementId, audioData);
      }
      
      // Update playback rate (this preserves pitch better than HTML5 audio)
      audioData.source.playbackRate.value = newRate;
      
    } catch (error) {
      // Fallback to standard HTML5 audio
      audioElement.playbackRate = newRate;
    }
  }
 
  // Stop audio playback
  stopAudio(elementId) {
    const audioData = this.audioSources.get(elementId);
    if (audioData) {
      try {
        audioData.source.disconnect();
        audioData.gainNode.disconnect();
      } catch (error) {
        // Silent error handling
      }
      this.audioSources.delete(elementId);
    }
  }
 
  // Update volume for existing audio
  updateVolume(elementId, newVolume) {
    const audioData = this.audioSources.get(elementId);
    if (audioData && audioData.gainNode) {
      audioData.gainNode.gain.value = newVolume;
    }
  }
 
  // Apply fade-in effect to audio element
  applyFadeIn(elementId, fadeInDuration = 1.0, targetVolume = 1.0) {
    const audioData = this.audioSources.get(elementId);
    if (!audioData || !audioData.gainNode) {
      return;
    }
 
    const { gainNode } = audioData;
    const currentTime = this.audioContext.currentTime;
    
    // Cancel any existing automation
    gainNode.gain.cancelScheduledValues(currentTime);
    
    // Start from 0 volume
    gainNode.gain.setValueAtTime(0, currentTime);
    
    // Fade in to target volume over the specified duration
    gainNode.gain.linearRampToValueAtTime(targetVolume, currentTime + fadeInDuration);
  }
 
  // Apply fade-out effect to audio element
  applyFadeOut(elementId, fadeOutDuration = 1.0, startTime = null) {
    const audioData = this.audioSources.get(elementId);
    if (!audioData || !audioData.gainNode) return;
 
    const { gainNode, audioElement } = audioData;
    const currentTime = this.audioContext.currentTime;
    const audioDuration = audioElement.duration;
    
    // If no start time specified, calculate from audio duration minus fade duration
    const fadeStartTime = startTime !== null ? startTime : Math.max(0, audioDuration - fadeOutDuration);
    const absoluteFadeStartTime = currentTime + fadeStartTime;
    
    // Cancel any existing automation after the fade start time
    gainNode.gain.cancelScheduledValues(absoluteFadeStartTime);
    
    // Set current volume at fade start time
    const currentVolume = gainNode.gain.value;
    gainNode.gain.setValueAtTime(currentVolume, absoluteFadeStartTime);
    
    // Fade out to 0 volume
    gainNode.gain.linearRampToValueAtTime(0, absoluteFadeStartTime + fadeOutDuration);
  }
 
  // Apply both fade-in and fade-out effects
  applyFadeEffects(elementId, fadeInDuration = 0, fadeOutDuration = 0, targetVolume = 1.0) {
    const audioData = this.audioSources.get(elementId);
    if (!audioData || !audioData.gainNode) {
      return;
    }
 
    const { gainNode, audioElement } = audioData;
    const currentTime = this.audioContext.currentTime;
    const audioDuration = audioElement.duration;
    
    // Cancel any existing automation
    gainNode.gain.cancelScheduledValues(currentTime);
    
    if (fadeInDuration > 0) {
      // Start from 0 volume
      gainNode.gain.setValueAtTime(0, currentTime);
      // Fade in to target volume
      gainNode.gain.linearRampToValueAtTime(targetVolume, currentTime + fadeInDuration);
      
      if (fadeOutDuration > 0) {
        // Hold target volume until fade-out starts
        const fadeOutStartTime = Math.max(fadeInDuration, audioDuration - fadeOutDuration);
        gainNode.gain.setValueAtTime(targetVolume, currentTime + fadeOutStartTime);
        // Fade out to 0
        gainNode.gain.linearRampToValueAtTime(0, currentTime + fadeOutStartTime + fadeOutDuration);
      }
    } else if (fadeOutDuration > 0) {
      // Only fade-out, start from current volume
      const fadeOutStartTime = Math.max(0, audioDuration - fadeOutDuration);
      gainNode.gain.setValueAtTime(targetVolume, currentTime + fadeOutStartTime);
      gainNode.gain.linearRampToValueAtTime(0, currentTime + fadeOutStartTime + fadeOutDuration);
    } else {
      // No fades, just set target volume
      gainNode.gain.setValueAtTime(targetVolume, currentTime);
    }
  }
 
  // Clear all fade effects and set volume immediately
  clearFadeEffects(elementId, volume = 1.0) {
    const audioData = this.audioSources.get(elementId);
    if (!audioData || !audioData.gainNode) {
      return;
    }
 
    const { gainNode } = audioData;
    const currentTime = this.audioContext.currentTime;
    
    // Cancel all scheduled changes
    gainNode.gain.cancelScheduledValues(currentTime);
    // Set volume immediately
    gainNode.gain.setValueAtTime(volume, currentTime);
  }
 
  // Apply noise reduction to audio element
  applyNoiseReduction(elementId, enabled = true) {
    const audioData = this.audioSources.get(elementId);
    if (!audioData) {
      return;
    }
 
    try {
      // The node that feeds into NR (or gainNode) is either the pitch
      // chain's output (merger) or the raw source when no pitch is active.
      const chainInput = audioData.pitchNodes
        ? audioData.pitchNodes.merger
        : audioData.source;
 
      if (enabled && !audioData.noiseReductionNodes) {
        // Create noise reduction filter chain
        const highpassFilter = this.audioContext.createBiquadFilter();
        const lowpassFilter = this.audioContext.createBiquadFilter();
        const compressor = this.audioContext.createDynamicsCompressor();
        
        highpassFilter.type = 'highpass';
        highpassFilter.frequency.setValueAtTime(80, this.audioContext.currentTime);
        highpassFilter.Q.setValueAtTime(0.7, this.audioContext.currentTime);
        
        lowpassFilter.type = 'lowpass';
        lowpassFilter.frequency.setValueAtTime(8000, this.audioContext.currentTime);
        lowpassFilter.Q.setValueAtTime(0.7, this.audioContext.currentTime);
        
        compressor.threshold.setValueAtTime(-24, this.audioContext.currentTime);
        compressor.knee.setValueAtTime(30, this.audioContext.currentTime);
        compressor.ratio.setValueAtTime(12, this.audioContext.currentTime);
        compressor.attack.setValueAtTime(0.003, this.audioContext.currentTime);
        compressor.release.setValueAtTime(0.25, this.audioContext.currentTime);
        
        audioData.noiseReductionNodes = {
          highpassFilter,
          lowpassFilter,
          compressor
        };
        
        // Insert NR chain between chainInput and gainNode
        try { chainInput.disconnect(audioData.gainNode); } catch (_) {}
        chainInput.connect(highpassFilter);
        highpassFilter.connect(lowpassFilter);
        lowpassFilter.connect(compressor);
        compressor.connect(audioData.gainNode);
        
      } else if (!enabled && audioData.noiseReductionNodes) {
        const { highpassFilter, lowpassFilter, compressor } = audioData.noiseReductionNodes;
        
        // Disconnect NR chain
        try { chainInput.disconnect(highpassFilter); } catch (_) {}
        highpassFilter.disconnect();
        lowpassFilter.disconnect();
        compressor.disconnect();
        
        // Reconnect chainInput directly to gainNode
        chainInput.connect(audioData.gainNode);
        
        delete audioData.noiseReductionNodes;
      }
      
    } catch (error) {
      // Silent error handling
    }
  }
 
  // Apply pitch shift to an audio/video element using delay-line modulation.
  // `pitchValue` is in the range -100..100 (0 = no shift).
  // Apply pitch shift to an audio/video element using perfectly smooth
  // looped AudioBuffer nodes to prevent "clicking" or "ticking" artifacts
  // that occur with standard Web Audio band-limited oscillators.
  applyPitchShift(elementId, pitchValue = 0) {
    const audioData = this.audioSources.get(elementId);
    if (!audioData || !this.audioContext) return;
 
    // --- Remove existing pitch nodes if present ---
    if (audioData.pitchNodes) {
      try {
        const p = audioData.pitchNodes;
        ['mod1', 'mod2', 'fadeMod1', 'fadeMod2', 'lfo1', 'lfo2', 'fadeIn1', 'fadeIn2'].forEach(name => {
          if (p[name] && typeof p[name].stop === 'function') p[name].stop();
        });
        Object.values(p).forEach(node => {
          if (node && typeof node.disconnect === 'function') node.disconnect();
        });
      } catch (_) { /* ignore cleanup errors */ }
      delete audioData.pitchNodes;
    }
 
    // --- Rebuild the audio chain from source ---
    try { audioData.source.disconnect(); } catch (_) {}
 
    if (pitchValue === 0) {
      this._connectSourceToOutput(audioData);
      return;
    }
 
    const ctx = this.audioContext;
    const now = ctx.currentTime;
 
    // Map -100..100 → semitones → loop speeds
    const semitones = (pitchValue / 100) * 12;
    const grainSize = 0.05; // 50 ms grain
    const pitchRatio = Math.pow(2, semitones / 12);
    const lfoFreq = Math.abs(pitchRatio - 1) / grainSize;
    const isUp = semitones > 0;
 
    // Clamp very low LFO frequencies to prevent division by zero / glitches
    const safeLfoFreq = Math.max(lfoFreq, 0.1);
 
    // Initialize mathematically perfect window buffers once
    if (!this.pitchBuffers) {
      const sr = ctx.sampleRate || 44100;
      const bufLen = sr; // 1 second length
      const fb = ctx.createBuffer(1, bufLen, sr);
      const dbUp = ctx.createBuffer(1, bufLen, sr);
      const dbDown = ctx.createBuffer(1, bufLen, sr);
      
      const fData = fb.getChannelData(0);
      const dUpData = dbUp.getChannelData(0);
      const dDownData = dbDown.getChannelData(0);
      
      for(let i = 0; i < bufLen; i++) {
        const x = i / bufLen;
        // Perfect smooth Hanning window
        fData[i] = 0.5 * (1 - Math.cos(2 * Math.PI * x));
        // Up sweep: 1.0 down to 0.0
        dUpData[i] = 1.0 - x;
        // Down sweep: 0.0 up to 1.0
        dDownData[i] = x;
      }
      this.pitchBuffers = { fade: fb, up: dbUp, down: dbDown };
    }
 
    // Delay lines - driven entirely by the mod buffers
    const delay1 = ctx.createDelay(1.0);
    const delay2 = ctx.createDelay(1.0);
    delay1.delayTime.value = 0;
    delay2.delayTime.value = 0;
 
    const mod1Gain = ctx.createGain();
    const mod2Gain = ctx.createGain();
    mod1Gain.gain.value = grainSize;
    mod2Gain.gain.value = grainSize;
 
    const mod1 = ctx.createBufferSource();
    const mod2 = ctx.createBufferSource();
    mod1.buffer = isUp ? this.pitchBuffers.up : this.pitchBuffers.down;
    mod2.buffer = isUp ? this.pitchBuffers.up : this.pitchBuffers.down;
    mod1.loop = true;
    mod2.loop = true;
    mod1.playbackRate.value = safeLfoFreq;
    mod2.playbackRate.value = safeLfoFreq;
 
    mod1.connect(mod1Gain);
    mod2.connect(mod2Gain);
    mod1Gain.connect(delay1.delayTime);
    mod2Gain.connect(delay2.delayTime);
 
    // Fade control - driven entirely by the fade buffers
    const fade1Gain = ctx.createGain();
    const fade2Gain = ctx.createGain();
    fade1Gain.gain.value = 0;
    fade2Gain.gain.value = 0;
 
    const fadeMod1 = ctx.createBufferSource();
    const fadeMod2 = ctx.createBufferSource();
    fadeMod1.buffer = this.pitchBuffers.fade;
    fadeMod2.buffer = this.pitchBuffers.fade;
    fadeMod1.loop = true;
    fadeMod2.loop = true;
    fadeMod1.playbackRate.value = safeLfoFreq;
    fadeMod2.playbackRate.value = safeLfoFreq;
 
    fadeMod1.connect(fade1Gain.gain);
    fadeMod2.connect(fade2Gain.gain);
 
    const merger = ctx.createGain();
    
    // Connect original source to delays -> gains -> merger
    audioData.source.connect(delay1);
    audioData.source.connect(delay2);
    delay1.connect(fade1Gain);
    delay2.connect(fade2Gain);
    fade1Gain.connect(merger);
    fade2Gain.connect(merger);
 
    // Re-link output
    if (audioData.noiseReductionNodes) {
      merger.connect(audioData.noiseReductionNodes.highpassFilter);
    } else {
      merger.connect(audioData.gainNode);
    }
 
    // Start playback: half-phase offset is exactly 0.5s into a 1s buffer
    mod1.start(now, 0);
    fadeMod1.start(now, 0);
    mod2.start(now, 0.5);
    fadeMod2.start(now, 0.5);
 
    // Store references for cleanup
    audioData.pitchNodes = {
      mod1, mod2, fadeMod1, fadeMod2,
      mod1Gain, mod2Gain, fade1Gain, fade2Gain,
      delay1, delay2, merger,
    };
  }
 
  // Helper: reconnect source → output respecting noiseReduction chain
  _connectSourceToOutput(audioData) {
    if (audioData.noiseReductionNodes) {
      audioData.source.connect(audioData.noiseReductionNodes.highpassFilter);
    } else {
      audioData.source.connect(audioData.gainNode);
    }
  }
 
  // Initialize an audio element in the Web Audio API context
  async initializeAudioElement(audioElement) {
    if (!audioElement || !audioElement.src) {
      return false;
    }
 
    try {
      // Initialize AudioContext if needed
      if (!this.isInitialized) {
        await this.initialize();
      }
 
      const elementId = audioElement.id || audioElement.src;
      
      // Check if already initialized
      if (this.audioSources.has(elementId)) {
        return true;
      }
 
      // Create Web Audio API nodes
      const source = this.audioContext.createMediaElementSource(audioElement);
      const gainNode = this.audioContext.createGain();
      
      // Connect the nodes
      source.connect(gainNode);
      gainNode.connect(this.audioContext.destination);
      
      // Store reference
      const audioData = { source, gainNode, audioElement };
      this.audioSources.set(elementId, audioData);
      
      return true;
      
    } catch (error) {
      return false;
    }
  }
 
  // Check if Web Audio API is supported
  static isSupported() {
    try {
      return !!(window.AudioContext || window.webkitAudioContext);
    } catch (error) {
      return false;
    }
  }
 
  // Clean up resources
  cleanup() {
    this.audioSources.forEach((audioData, elementId) => {
      this.stopAudio(elementId);
    });
    
    if (this.audioContext && this.audioContext.state !== 'closed') {
      this.audioContext.close();
    }
    
    this.isInitialized = false;
  }
}

const audioProcessor = new AudioProcessor();
 
export default audioProcessor;

/**
 * Check if a fabric object's visual state has changed since last capture.
 * Returns false if we can skip the expensive captureFabricObjectState + texImage2D.
 *
 * For video objects: only treat as "changed" if the underlying <video> is
 * actually playing. During GL transitions one of the two source videos is
 * almost always paused (outside its own timeframe), so re-capturing its
 * frame every 50 ms is pure waste — the pixels are identical. This single
 * check eliminates ~50% of capture+upload cost during a typical transition.
 */
function hasVisualStateChanged(fabricObject, lastState, isVideo, element) {
  Eif (!lastState) return true; // first frame → always update
  if (lastState.fabricObject !== fabricObject) return true; // fabric object recreated → always update
 
  if (isVideo) {
    // CEF headless path: __applyVideoFrame swaps fab.__cefBitmap each frame
    // and leaves the <video> DOM element untouched, so currentTime never
    // advances. Detect a fresh bitmap by identity comparison instead.
    if (fabricObject.__cefBitmap) {
      if (lastState.cefBitmap !== fabricObject.__cefBitmap) return true;
      // Bitmap unchanged → fall through to transform-only diff below.
      const opacityDelta = Math.abs((fabricObject.opacity ?? 1) - (lastState.opacity ?? 1));
      const scaleXDelta = Math.abs((fabricObject.scaleX ?? 1) - (lastState.scaleX ?? 1));
      const scaleYDelta = Math.abs((fabricObject.scaleY ?? 1) - (lastState.scaleY ?? 1));
      const angleDelta = Math.abs((fabricObject.angle || 0) - (lastState.angle || 0));
      return opacityDelta > 0.001 || scaleXDelta > 0.003 || scaleYDelta > 0.003 || angleDelta > 0.1;
    }
 
    // Get the underlying <video> DOM element to check its play state.
    const videoEl =
      (element && element.properties && document.getElementById(element.properties.elementId)) ||
      fabricObject._element;
    if (videoEl && typeof videoEl.paused === 'boolean') {
      // If the video is paused, its current frame won't change — capture once
      // and skip until it plays again. Track the last-captured currentTime so
      // we still re-capture when the video is seeked while paused.
      const curT = videoEl.currentTime || 0;
      if (videoEl.paused) {
        if (
          lastState.videoCurrentTime != null &&
          Math.abs(curT - lastState.videoCurrentTime) < 0.001
        ) {
          // Still need to detect transform changes (scale/opacity/angle from effects)
          const opacityDelta = Math.abs((fabricObject.opacity ?? 1) - (lastState.opacity ?? 1));
          const scaleXDelta = Math.abs((fabricObject.scaleX ?? 1) - (lastState.scaleX ?? 1));
          const scaleYDelta = Math.abs((fabricObject.scaleY ?? 1) - (lastState.scaleY ?? 1));
          const angleDelta = Math.abs((fabricObject.angle || 0) - (lastState.angle || 0));
          return (
            opacityDelta > 0.001 || scaleXDelta > 0.003 || scaleYDelta > 0.003 || angleDelta > 0.1
          );
        }
      }
    }
    return true; // playing → frames change every tick
  }
 
  // For static images: only re-capture if opacity/scale/angle actually moved
  const opacityDelta = Math.abs((fabricObject.opacity ?? 1) - (lastState.opacity ?? 1));
  const scaleXDelta = Math.abs((fabricObject.scaleX ?? 1) - (lastState.scaleX ?? 1));
  const scaleYDelta = Math.abs((fabricObject.scaleY ?? 1) - (lastState.scaleY ?? 1));
  const angleDelta = Math.abs((fabricObject.angle || 0) - (lastState.angle || 0));
 
  return opacityDelta > 0.001 || scaleXDelta > 0.003 || scaleYDelta > 0.003 || angleDelta > 0.1;
}

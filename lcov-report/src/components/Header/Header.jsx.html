import React, { useState, useRef, useEffect } from 'react';
import styles from './Header.module.scss';
import { Navigation } from 'components/Header/Navigation/Navigation';
import { ButtonWithIcon } from 'components/reusableComponents/ButtonWithIcon';
import { useSelector, useDispatch } from 'react-redux';
import { user, logout } from '../../redux/auth';
import { NavLink, useLocation } from 'react-router-dom';
import { BsGear } from 'react-icons/bs';
import { MinimalHeader } from './MinimalHeader/MinimalHeader';
import { useAuth } from '@clerk/clerk-react'; // Import useAuth from Clerk
import videfyLogo from 'images/videfyLogo.png';
 
function Header() {
  const { isSignedIn, signOut } = useAuth(); // Access signOut from useAuth
  const { username } = useSelector(user);
  const dispatch = useDispatch();
  const location = useLocation();
  const [isScrolled, setIsScrolled] = useState(false);
  const [isDropdownOpen, setIsDropdownOpen] = useState(false);
  const [isMenuOpen, setIsMenuOpen] = useState(false);
  const avatarRef = useRef(null);
  const dropdownRef = useRef(null);
 
  const isVideoCreationPage = () => location.pathname.includes('createVideo');
  const isProjectCreationPage = () =>
    location.pathname.includes('createProject');
 
  const isStoryboardPage = () => location.pathname.includes('storyboard');
 
  const isHomePage = () => location.pathname.includes('home') || location.pathname.includes('gen-ai') || location.pathname.includes('gallery') || location.pathname.includes('my-items') || location.pathname.includes('aiAvatar');
  const isAskOurTeamPage = () => location.pathname.includes('ask-our-team');
  const shouldShowBorder = () => {
    return !isVideoCreationPage() || isScrolled;
  };
 
  useEffect(() => {
    const shouldUseMinimalHeader =
      isVideoCreationPage() || isProjectCreationPage() || isStoryboardPage();
    document.body.style.paddingTop = shouldUseMinimalHeader ? '0' : '72px';
 
    return () => {
      document.body.style.paddingTop = '72px';
    };
  }, [location.pathname]);
 
  useEffect(() => {
    const handleScroll = () => {
      if (window.scrollY > 50) {
        setIsScrolled(true);
      } else {
        setIsScrolled(false);
      }
    };
 
    window.addEventListener('scroll', handleScroll);
    return () => window.removeEventListener('scroll', handleScroll);
  }, []);
 
  const handleLogOut = async () => {
    await signOut(); // Sign out of Clerk first
    await dispatch(logout()); // Then logout from backend (removes token from DB)
    setIsDropdownOpen(false);
  };
 
  const handleMenuEnter = () => {
    setIsMenuOpen(true);
  };
 
  const handleMenuLeave = () => {
    setIsMenuOpen(false);
  };
 
  const handleDropdownEnter = () => {
    setIsDropdownOpen(true);
  };
 
  const handleDropdownLeave = () => {
    setIsDropdownOpen(false);
  };
 
  const getUserInitial = () => {
    return username ? username.charAt(0).toUpperCase() : '';
  };
 
  if (
    isVideoCreationPage() ||
    isProjectCreationPage() ||
    isStoryboardPage() ||
    isHomePage || isAskOurTeamPage()
  ) {
    return (
      <MinimalHeader
        isScrolled={isScrolled}
        isStoryboardPage={isStoryboardPage()}
        isHomePage={isHomePage()}
        isAskOurTeamPage={isAskOurTeamPage()}
      />
    );
  }
 
  return (
    <header className={isScrolled ? styles.scrolled : ''}>
      <div className={styles.header_box}>
        <NavLink to="/profile/home">
          <img className={styles.header_logo} src={videfyLogo} alt="Videfy Logo" />
          <h1 className={styles.header_text}>Videfy.ai</h1>
        </NavLink>
        <div className={styles.header_nav_box}>
          <Navigation />
        </div>
        <div className={styles.header_btn_set}>
          {isSignedIn ? (
            <>
              <ButtonWithIcon
                text="Create project"
                icon="PlusIcon"
                path="profile/createProject"
                classNameButton={styles.header_btn_bright}
                classNameIcon={styles.icon}
                color="white"
                accentColor="white"
              />
              <button
                ref={avatarRef}
                className={styles.header_avatar}
                onMouseEnter={handleDropdownEnter}
              >
                {getUserInitial()}
              </button>
              {isDropdownOpen && (
                <div
                  ref={dropdownRef}
                  className={`${styles.header_dropdown_user} ${
                    isDropdownOpen ? styles.visible : styles.hidden
                  }`}
                  onMouseEnter={handleDropdownEnter}
                  onMouseLeave={handleDropdownLeave}
                >
                  <div className={styles.header_dropdown_content}>
                    <div className={styles.header_user_info}>
                      <div className={styles.header_avatar}>
                        {getUserInitial()}
                      </div>
                      <span className={styles.user_name_dropdown}>
                        {username}
                      </span>
                    </div>
                    <button
                      className={styles.header_dropdown_button}
                      onClick={() => {
                        setIsDropdownOpen(false);
                        window.location.href = '/profile/settings';
                      }}
                    >
                      <BsGear size={20} />
                      Settings
                    </button>
                    <button
                      className={styles.header_dropdown_button}
                      onClick={handleLogOut}
                    >
                      <svg
                        width="20"
                        height="20"
                        viewBox="0 0 20 20"
                        fill="none"
                      >
                        <path
                          d="M7.5 17.5H4.16667C3.72464 17.5 3.30072 17.3244 2.98816 17.0118C2.67559 16.6993 2.5 16.2754 2.5 15.8333V4.16667C2.5 3.72464 2.67559 3.30072 2.98816 2.98816C3.30072 2.67559 3.72464 2.5 4.16667 2.5H7.5M13.3333 14.1667L17.5 10M17.5 10L13.3333 5.83333M17.5 10H7.5"
                          stroke="currentColor"
                          strokeWidth="1.66667"
                          strokeLinecap="round"
                          strokeLinejoin="round"
                        />
                      </svg>
                      Log out
                    </button>
                  </div>
                </div>
              )}
            </>
          ) : (
            <ButtonWithIcon
              text="Log in"
              path="/login"
              classNameButton={styles.header_btn}
              classNameIcon={styles.icon}
              color="white"
              accentColor="black"
            />
          )}
        </div>
      </div>
    </header>
  );
}
 
export { Header };

import React, { useState } from 'react';
import { createPortal } from 'react-dom';
import { SearchIcon, LikeLanguageIcon, LikeLanguageVoiceoverIcon } from 'components/Icons';
import styles from './CreateVoiceoverModal.module.scss';
 
const AccentDropdown = ({
  dropdownRef,
  buttonRef,
  selectedAccent,
  selectedLanguage,
  narratorSettings,
  accentSearchValue,
  showFavoritesOnly,
  likingAccentId,
  accents,
  getDropdownPosition,
  handleAccentSelect,
  handleAccentSearchChange,
  handleAccentSearchKeyDown,
  handleLikeAccent,
  setShowFavoritesOnly,
  getAccentsForLanguage,
}) => {
  const [hoveredAccent, setHoveredAccent] = useState(null);
  const [hoveredFavoritesButton, setHoveredFavoritesButton] = useState(false);
 
  const position = getDropdownPosition(buttonRef);
  
  // Don't render until position is calculated
  if (!position || position.left == null || (position.top == null && position.bottom == null)) {
    return null;
  }
 
  return createPortal(
    <div
      ref={dropdownRef}
      className={`${styles.portalDropdown} ${styles.accentDropdown}`}
      style={{
        position: 'fixed',
        left: position.left,
        zIndex: 100000,
        maxHeight: position.maxHeight ?? 320,
        overflowY: 'auto',
        height: 'auto',
        ...(position.bottom != null
          ? { top: 'auto', bottom: position.bottom }
          : { top: position.top, bottom: 'auto' }),
      }}
      onClick={(e) => e.stopPropagation()}
    >
      {/* <div className={styles.dropdownHeader}>
        Recently Used
      </div>
      {narratorSettings.recentAccent && (
        <button 
          className={`${styles.portalDropdownItem} ${styles.accent_recently_used_last} ${selectedAccent === narratorSettings.recentAccent.accent ? styles.selected : ''}`}
          onClick={(e) => {
            e.stopPropagation();
            handleAccentSelect(narratorSettings.recentAccent.accent);
          }}
        >
          <span>{narratorSettings.recentAccent.accent}</span>
        </button>
      )} */}
      <div className={styles.searchInputContainer}>
        <div className={styles.searchInputWrapper}>
          <SearchIcon size="14px" color="rgba(255, 255, 255, 0.32)" />
          <input
            type="text"
            placeholder="Search accent"
            className={styles.searchInput}
            value={accentSearchValue}
            onChange={handleAccentSearchChange}
            onKeyDown={handleAccentSearchKeyDown}
            maxLength={30}
          />
        </div>
        <div 
          className={styles.likeIconWrapper}
          onClick={(e) => {
            e.stopPropagation();
            setShowFavoritesOnly(!showFavoritesOnly);
          }}
          onMouseEnter={() => setHoveredFavoritesButton(true)}
          onMouseLeave={() => setHoveredFavoritesButton(false)}
        >
          <LikeLanguageIcon 
            size="18px" 
            color={
              showFavoritesOnly 
                ? "var(--accent-color)" 
                : hoveredFavoritesButton
                  ? "#FFF"
                  : "rgba(255, 255, 255, 0.4)"
            } 
          />
        </div>
      </div>
      
      {showFavoritesOnly && (
        <>
          {narratorSettings.favoriteAccents && narratorSettings.favoriteAccents.length > 0 ? (
            <>
              {narratorSettings.favoriteAccents
                .filter(favoriteAccent => {
                  const accentMatchesSearch = accentSearchValue === '' || 
                    favoriteAccent.accent.toLowerCase().includes(accentSearchValue.toLowerCase());
                  const accentMatchesLanguage = getAccentsForLanguage(accents, selectedLanguage)
                    .some(accent => accent.value === favoriteAccent.accent);
                  return accentMatchesSearch && accentMatchesLanguage;
                })
                .map((favoriteAccent) => (
                  <button 
                    key={favoriteAccent.accent}
                    className={`${styles.portalDropdownItem} ${styles.favoriteItem} ${selectedAccent === favoriteAccent.accent ? styles.selected : ''}`}
                    onClick={(e) => {
                      e.stopPropagation();
                      handleAccentSelect(favoriteAccent.accent);
                    }}
                  >
                    <span>{favoriteAccent.accent}</span>
                    <div 
                      onClick={(e) => {
                        e.stopPropagation();
                        handleLikeAccent(favoriteAccent.accent, e);
                      }}
                      onMouseEnter={() => setHoveredAccent(favoriteAccent.accent)}
                      onMouseLeave={() => setHoveredAccent(null)}
                    >
                      <LikeLanguageVoiceoverIcon 
                        size="14px" 
                        color={
                          likingAccentId === favoriteAccent.accent 
                            ? "rgba(255, 255, 255, 0.4)" 
                            : hoveredAccent === favoriteAccent.accent
                              ? "#FFF"
                              : "var(--accent-color)"
                        } 
                      />
                    </div>
                  </button>
                ))}
            </>
          ) : (
            <div className={styles.noFavoritesMessage}>
              No favorite accents for {selectedLanguage} yet
            </div>
          )}
        </>
      )}
      
      {!showFavoritesOnly && getAccentsForLanguage(accents, selectedLanguage)
        .filter(accent => {
          const matchesSearch = accentSearchValue === '' || 
            accent.name.toLowerCase().includes(accentSearchValue.toLowerCase()) ||
            accent.value.toLowerCase().includes(accentSearchValue.toLowerCase());
          
          return matchesSearch;
        })
        .map((accent) => (
        <button 
          key={accent.value}
          className={`${styles.portalDropdownItem} ${selectedAccent === accent.value ? styles.selected : ''}`}
          onClick={(e) => {
            e.stopPropagation();
            handleAccentSelect(accent.value);
          }}
        >
          <span>{accent.name}</span>
          <div 
            onClick={(e) => {
              e.stopPropagation();
              handleLikeAccent(accent.value, e);
            }}
            onMouseEnter={() => setHoveredAccent(accent.value)}
            onMouseLeave={() => setHoveredAccent(null)}
          >
            <LikeLanguageVoiceoverIcon 
              size="14px" 
              color={
                likingAccentId === accent.value
                  ? "rgba(255, 255, 255, 0.2)"
                  : narratorSettings.favoriteAccents && 
                    narratorSettings.favoriteAccents.some(fav => fav.accent === accent.value)
                    ? 'var(--accent-color)' 
                    : hoveredAccent === accent.value
                      ? '#FFF'
                      : 'rgba(255, 255, 255, 0.4)'
              }
            />
          </div>
        </button>
      ))}
    </div>,
    document.body
  );
};
 
export default AccentDropdown;

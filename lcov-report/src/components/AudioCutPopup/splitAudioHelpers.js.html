/**
 * Helper functions for managing split audio parts
 * Supports unlimited number of parts instead of hardcoded 8
 */
 
/**
 * Generate unique ID for a split part
 */
export const generatePartId = (index = null) => {
  if (index !== null) {
    return `part-${index}`;
  }
  return `part-${Date.now()}-${Math.random().toString(36).substring(2, 9)}`;
};
 
/**
 * Convert legacy part keys to new ID format
 * For backward compatibility
 */
export const normalizePartKey = (key) => {
  const legacyMap = {
    'part1': 'part-0',
    'part1a': 'part-1',
    'part1b': 'part-2',
    'part1c': 'part-3',
    'part2a': 'part-4',
    'part2b': 'part-5',
    'part2c': 'part-6',
    'part2': 'part-7',
  };
  return legacyMap[key] || key;
};
 
/**
 * Convert splitAudioFiles object to array format
 * @param {Object} splitAudioFiles - Object with part keys
 * @returns {Array} Array of parts with id and index
 */
export const splitFilesToArray = (splitAudioFiles) => {
  if (!splitAudioFiles || typeof splitAudioFiles !== 'object') {
    return [];
  }
 
  // If already in array format, return as is
  if (Array.isArray(splitAudioFiles)) {
    return splitAudioFiles.map((part, index) => ({
      ...part,
      id: part.id || generatePartId(index),
      index: part.index !== undefined ? part.index : index,
    }));
  }
 
  const parts = [];
  const processedKeys = new Set();
  
  Object.keys(splitAudioFiles).forEach((key) => {
    const part = splitAudioFiles[key];
    if (part) {
      parts.push({
        ...part,
        id: part.id || generatePartId(parts.length),
        originalKey: key,
        index: part.index !== undefined ? part.index : parts.length,
      });
      processedKeys.add(key);
    }
  });
 
  const hasIndices = parts.some(p => p.index !== undefined);
  if (hasIndices) {
    return parts.sort((a, b) => {
      if (a.index !== undefined && b.index !== undefined) {
        return a.index - b.index;
      }
      if (a.index !== undefined) return -1;
      if (b.index !== undefined) return 1;
      return (a.originalKey || a.id || '').localeCompare(b.originalKey || b.id || '');
    });
  }
 
  const orderedParts = parts.map((part, index) => ({
    ...part,
    index: part.index !== undefined ? part.index : index,
  }));
 
  return orderedParts.sort((a, b) => {
    if (a.index !== undefined && b.index !== undefined) {
      return a.index - b.index;
    }
    return 0;
  });
};
 
/**
 * Convert array format back to object format (for backward compatibility)
 * @param {Array} partsArray - Array of parts
 * @returns {Object} Object with part keys (originalKey if available, otherwise id)
 */
export const arrayToSplitFiles = (partsArray) => {
  if (!Array.isArray(partsArray)) {
    return partsArray || {};
  }
 
  const result = {};
  partsArray.forEach((part) => {
    // Use originalKey for backward compatibility with UI, fallback to id
    const key = part.originalKey || part.id;
    if (key) {
      // Remove helper fields but keep id for tracking
      const { originalKey, index, ...partData } = part;
      result[key] = {
        ...partData,
        id: part.id, // Keep id for tracking optimistic updates
      };
    }
  });
  return result;
};
 
/**
 * Get next part ID when splitting
 * @param {string} currentPartId - ID of part being split
 * @param {Array} existingParts - Array of existing parts
 * @returns {Object} Object with leftPartId and rightPartId
 */
export const getSplitPartIds = (currentPartId, existingParts) => {
  const currentIndex = existingParts.findIndex(p => p.id === currentPartId);
  
  if (currentIndex === -1) {
    // If part not found, generate new IDs
    const newIndex = existingParts.length;
    return {
      leftPartId: generatePartId(newIndex),
      rightPartId: generatePartId(newIndex + 1),
    };
  }
 
  // Insert new part after current one
  const leftPartId = currentPartId;
  const rightPartId = generatePartId(currentIndex + 1);
 
  return { leftPartId, rightPartId };
};
 
/**
 * Insert new parts after splitting
 * @param {Array} existingParts - Current parts array
 * @param {string} partToSplitId - ID of part being split
 * @param {Object} leftPart - Left part data
 * @param {Object} rightPart - Right part data
 * @returns {Array} New parts array
 */
export const insertSplitParts = (existingParts, partToSplitId, leftPart, rightPart) => {
  const currentIndex = existingParts.findIndex(p => p.id === partToSplitId);
  
  if (currentIndex === -1) {
    // If part not found, just add both parts
    return [...existingParts, leftPart, rightPart];
  }
 
  // Replace current part with left part, insert right part after
  const newParts = [...existingParts];
  newParts[currentIndex] = { ...leftPart, id: partToSplitId, index: currentIndex };
  
  // Update indices for remaining parts
  for (let i = currentIndex + 1; i < newParts.length; i++) {
    newParts[i] = { ...newParts[i], index: i + 1 };
  }
  
  // Insert right part
  newParts.splice(currentIndex + 1, 0, { ...rightPart, index: currentIndex + 1 });
  
  return newParts;
};
 
/**
 * Get ordered parts for rendering
 * Maintains chronological order based on original audio timeline
 */
export const getOrderedParts = (partsArray) => {
  if (!Array.isArray(partsArray)) {
    return [];
  }
  
  // Sort by index if available, otherwise by id
  return [...partsArray].sort((a, b) => {
    if (a.index !== undefined && b.index !== undefined) {
      return a.index - b.index;
    }
    return (a.id || '').localeCompare(b.id || '');
  });
};
 
/**
 * Find part by ID
 */
export const findPartById = (partsArray, partId) => {
  if (!Array.isArray(partsArray)) {
    return null;
  }
  return partsArray.find(p => p.id === partId);
};
 
/**
 * Get first available part
 */
export const getFirstAvailablePart = (partsArray) => {
  if (!Array.isArray(partsArray) || partsArray.length === 0) {
    return null;
  }
  return partsArray.find(p => p.url) || partsArray[0];
};
 
/**
 * Get next part keys for split operation
 */
export const getNextPartKeys = (currentKey, allParts = []) => {
  const timestamp = Date.now();
  const random = Math.random().toString(36).substring(2, 9);
  const newRightKey = `part-${timestamp}-${random}`;
  
  return {
    left: currentKey,
    right: newRightKey,
  };
};

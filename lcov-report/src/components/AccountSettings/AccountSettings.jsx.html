import { ButtonWithIcon } from 'components/reusableComponents/ButtonWithIcon';
import styles from './AccountSettings.module.scss';
import { useSelector, useDispatch } from 'react-redux';
import { user, update } from '../../redux/auth';
import { store as reduxStore } from '../../redux/store';
import { useState, useEffect } from 'react';
import axios from 'axios';
import FormUpdateUser from 'components/FormUpdateUser/FormUpdateUser';
import { ThemeSelector } from 'components/ThemeSelector/ThemeSelector';
import { AddImageIcon, FilterCheckBoxIcon } from 'components/Icons';
import PasswordChangeModal from 'components/PasswordChangeModal/PasswordChangeModal';
import { useAvatar } from '../../contexts/AvatarContext';
import PasswordToggleButton from 'components/PasswordToggleButton/PasswordToggleButton';
import GeometricCopyButton from 'components/reusableComponents/GeometricCopyButton/GeometricCopyButton';
import formStyles from 'components/FormUpdateUser/FormUpdateUser.module.scss';
 
const AccountSettings = ({ onUserUpdate }) => {
  const [localUserName, setLocalUserName] = useState('');
  const [notificationFilters, setNotificationFilters] = useState(['All']);
  const [showPasswordForm, setShowPasswordForm] = useState(false);
  // const [showPasswordModal, setShowPasswordModal] = useState(false);
  const [showPasswordChangeModal, setShowPasswordChangeModal] = useState(false);
  const { avatarUrl, updateAvatar } = useAvatar();
  const { username, email } = useSelector(user);
  const dispatch = useDispatch();
  const currentUser = useSelector(user);
  const [integrationsToken, setintegrationsToken] = useState('');
  const [isTokenVisible, setIsTokenVisible] = useState(false);
  const [isFetchingToken, setIsFetchingToken] = useState(false);
 
  useEffect(() => {
    setLocalUserName(username);
  }, [username]);
 
 
  const updateUsername = (newUsername) => {
    if (newUsername !== currentUser.username) {
      dispatch(
        update({
          email: currentUser.email,
          newUsername,
        })
      );
    }
  };
 
  const onChangeUserName = e => {
    const newUsername = e.target.value;
    setLocalUserName(newUsername);
  };
 
  const handleKeyDown = e => {
    if (e.key === 'Enter') {
      e.target.blur();
    }
  };
 
  const handleBlur = () => {
    updateUsername(localUserName);
  };
 
  const handleNotificationFilterChange = (filter) => {
    if (filter === 'All') {
      setNotificationFilters(['All']);
    } else {
      const newFilters = notificationFilters.includes('All')
        ? [filter]
        : notificationFilters.includes(filter)
          ? notificationFilters.filter(f => f !== filter)
          : [...notificationFilters, filter];
 
      if (newFilters.length === 0) {
        setNotificationFilters(['All']);
      } else {
        setNotificationFilters(newFilters);
      }
    }
  };
 
 
 
  const handlePasswordChangeModalClose = () => {
    setShowPasswordChangeModal(false);
  };
 
  const handlePasswordChangeModalSuccess = () => {
    setShowPasswordChangeModal(false);
  };
 
  const handlePasswordChange = () => {
    setShowPasswordChangeModal(true);
  };
 
  const handleImageUpload = event => {
    const file = event.target.files[0];
    if (!file) return;
 
    const url = URL.createObjectURL(file);
    updateAvatar(url);
  };
 
  const handleAvatarClick = () => {
    document.getElementById('avatar-upload').click();
  };
 
  const handleToggleTokenVisibility = () => {
    setIsTokenVisible(prev => !prev);
  };
 
  const handleCopyToken = async () => {
    try {
      await navigator.clipboard.writeText(integrationsToken || '');
    } catch (e) {
      // ignore copy errors silently
    }
  };
 
  const handleGetIntegrationsToken = async () => {
    try {
      setIsFetchingToken(true);
      const res = await axios.get('/auth/integrations/token');
      setintegrationsToken(res?.data?.token || '');
    } catch (e) {
      // swallow error; optionally show toast in future
    } finally {
      setIsFetchingToken(false);
    }
  };
 
  return (
    <section className={styles.container}>
      <div className={styles.header_top}>
        <div className={styles.avatar} onClick={handleAvatarClick}>
          <input
            id="avatar-upload"
            type="file"
            accept="image/*"
            onChange={handleImageUpload}
            style={{ display: 'none' }}
          />
          {avatarUrl ? (
            <img
              src={avatarUrl}
              alt="Selected"
              className={styles.user_image}
              loading="lazy"
            />
          ) : (
            <div className={styles.imagePlaceholder}>
              <AddImageIcon size="26px" color="#FFFFFF99" />
            </div>
          )}
        </div>
        <div className={styles.user_data}>
          <h1>@{localUserName}</h1>
          <p>{email}</p>
        </div>
      </div>
      <div className={styles.settings_content}>
        <div className={styles.account_settings_box}>
          <div className={styles.input_container}>
            <h2>Name</h2>
            <input
              type="text"
              placeholder="Name Surname"
              value={localUserName}
              onChange={onChangeUserName}
              onBlur={handleBlur}
              onKeyDown={handleKeyDown}
            />
          </div>
          {/* <div className={styles.email_box}> */}
          {/* <h2>Email</h2> */}
          {/* <div className={styles.email_item}>
              <div className={styles.email_info}>
                <ButtonWithIcon
                  icon="EmailIcon"
                  color="var(--accent-color)"
                  classNameButton={styles.email_btn}
                />
                <div className={styles.email_text}>
                  <h3>{email}</h3>
                  <p>1 month ago</p>
                </div>
              </div> */}
 
          {/* <button className={styles.submit_btn}>Change</button> */}
          {/* </div> */}
          {/* </div> */}
          <div className={styles.theme_container}>
            <p className={styles.theme_title}>System</p>
            <div className={styles.notification_section}>
              <h2>Notification channel</h2>
              <ul className={styles.notification_filters}>
                {['All', 'Email', 'Push', 'SMS', 'Slack'].map(filter => (
                  <li className={styles.filter_item} key={filter}>
                    <label className={`${styles.filter_label} ${notificationFilters.includes(filter) ? styles.selected : ''}`}>
                      {notificationFilters.includes(filter) ? (
                        <span className={styles.filter_checked}>
                          <FilterCheckBoxIcon />
                        </span>
                      ) : (
                        <span className={styles.filter_checked}></span>
                      )}
                      <input
                        className={styles.check_input}
                        type="checkbox"
                        id={filter}
                        name={filter}
                        value={filter}
                        checked={notificationFilters.includes(filter)}
                        onChange={() => handleNotificationFilterChange(filter)}
                      />
                      {filter}
                    </label>
                  </li>
                ))}
              </ul>
            </div>
            <div className={styles.theme_color_section}>
              <h2>Theme color</h2>
              <ThemeSelector />
            </div>
          </div>
        </div>
        <div className={styles.email_settings_box}>
          <FormUpdateUser onPasswordChange={handlePasswordChange} />
        </div>
        <div className={styles.tokens_container}>
          <p className={styles.tokens_title}>Tokens</p>
          {!integrationsToken ? (
            <button className={styles.submit_btn} onClick={handleGetIntegrationsToken} disabled={isFetchingToken}>
              {isFetchingToken ? 'Generating…' : 'Get token for Adobe Premiere Pro / Davinci Resolve'}
            </button>
          ) : (
            <div className={styles.token_container}>
              <div className={`${styles.token_text} ${isTokenVisible ? styles.token_text_visible : styles.token_text_hidden}`} title={isTokenVisible ? integrationsToken : ''}>
                {isTokenVisible ? (integrationsToken || '') : '********************************'}
              </div>
              <div className={styles.token_actions}>
                <div className={styles.eye_wrapper}>
                  <PasswordToggleButton
                    isPasswordShown={isTokenVisible}
                    togglePasswordVisibility={handleToggleTokenVisibility}
                  />
                </div>
                <button className={styles.copy_btn} onClick={handleCopyToken} aria-label="Copy token">Copy</button>
              </div>
            </div>
          )}
        </div>
      </div>
 
      {showPasswordForm && (
        <div className={styles.password_form_container}>
          <FormUpdateUser />
        </div>
      )}
 
      {/* 
      <ChangePasswordModal
        isOpen={showPasswordModal}
        onClose={handlePasswordModalClose}
        onSuccess={handlePasswordModalSuccess}
      />
      */}
 
      <PasswordChangeModal
        isOpen={showPasswordChangeModal}
        onClose={handlePasswordChangeModalClose}
        onSuccess={handlePasswordChangeModalSuccess}
      />
    </section>
  );
};
export { AccountSettings };

const validateSearchInput = (value) => {
  const allowedChars = /^[a-zA-Z\s\-']*$/;
  return allowedChars.test(value);

const PORTAL_MIN_WIDTH = 196;
const PORTAL_MAX_HEIGHT = 304;
/** Space between `languageDropdown__toggle` bottom and portal top. */
const PORTAL_GAP = 2;
 
function computePortalLayout(rect) {
  const vw = window.innerWidth;
  const vh = window.innerHeight;
  const width = Math.min(
    Math.max(PORTAL_MIN_WIDTH, rect.width),
    Math.max(PORTAL_MIN_WIDTH, vw - 16),
  );
  let left = rect.left;
  if (left + width > vw - 8) {
    left = Math.max(8, vw - width - 8);
  }
  let top = rect.bottom + PORTAL_GAP;
  if (top + PORTAL_MAX_HEIGHT > vh - 8 && rect.top > PORTAL_MAX_HEIGHT + 8) {
    top = Math.max(8, rect.top - PORTAL_MAX_HEIGHT - PORTAL_GAP);
  }
  return { top, left, width };
}
 
const SubtitlesLanguageDropdown = observer(
  ({ selectedLanguage: selectedLanguageProp, onLanguageChange }) => {
    const { data: narratorSettingsData } = useGetNarratorSettingsQuery();
    const [updateRecentLanguage] = useUpdateRecentLanguageMutation();
    const [toggleFavoriteLanguage] = useToggleFavoriteLanguageMutation();
 
    const [isDropdownOpen, setDropdownOpen] = useState(false);
    const [localSelectedLanguage, setLocalSelectedLanguage] = useState('English');
    const [languageSearchValue, setLanguageSearchValue] = useState('');
    const [showFavoritesOnly, setShowFavoritesOnly] = useState(false);
    const [likingLanguageId, setLikingLanguageId] = useState(null);
    const [narratorSettings, setNarratorSettings] = useState({
      recentLanguages: [],
      favoriteLanguages: [],
    });
 
    const toggleRef = useRef(null);
    const portalRef = useRef(null);
    const [portalBox, setPortalBox] = useState({
      top: 0,
      left: 0,
      width: PORTAL_MIN_WIDTH,
    });
 
    const updatePortalPosition = useCallback(() => {
      const el = toggleRef.current;
      if (!el) return;
      setPortalBox(computePortalLayout(el.getBoundingClientRect()));
    }, []);
 
    useEffect(() => {
      if (narratorSettingsData) {
        const settings = narratorSettingsData?.data || {
          recentLanguages: [],
          favoriteLanguages: [],
        };
        setNarratorSettings(settings);
      }
    }, [narratorSettingsData]);
 
    useEffect(() => {
      if (!isDropdownOpen) return;
      const handleMouseDown = (event) => {
        const t = toggleRef.current;
        const p = portalRef.current;
        if (t?.contains(event.target) || p?.contains(event.target)) {
          return;
        }
        setDropdownOpen(false);
        setShowFavoritesOnly(false);
        setLanguageSearchValue('');
      };
      document.addEventListener('mousedown', handleMouseDown);
      return () => document.removeEventListener('mousedown', handleMouseDown);
    }, [isDropdownOpen]);
 
    useLayoutEffect(() => {
      if (!isDropdownOpen) return;
      updatePortalPosition();
      const id = requestAnimationFrame(() => {
        updatePortalPosition();
      });
      return () => cancelAnimationFrame(id);
    }, [isDropdownOpen, updatePortalPosition]);
 
    useEffect(() => {
      if (!isDropdownOpen) return;
      const ro = new ResizeObserver(() => updatePortalPosition());
      const el = toggleRef.current;
      if (el) ro.observe(el);
      window.addEventListener('scroll', updatePortalPosition, true);
      window.addEventListener('resize', updatePortalPosition);
      return () => {
        ro.disconnect();
        window.removeEventListener('scroll', updatePortalPosition, true);
        window.removeEventListener('resize', updatePortalPosition);
      };
    }, [isDropdownOpen, updatePortalPosition]);
 
    const selectedLanguage = selectedLanguageProp ?? localSelectedLanguage;
 
    const selectedLangMeta = languages.find((l) => l.value === selectedLanguage) || languages[0];
 
    const handleLanguageSearchChange = (e) => {
      const value = e.target.value;
      if (validateSearchInput(value) && value.length <= 30) {
        setLanguageSearchValue(value);
      }
    };
 
    const handleLanguageSearchKeyDown = (e) => {
      if (e.key >= '0' && e.key <= '9') {
        e.preventDefault();
      }
    };
 
    const handleLanguageSelect = async (language) => {
      if (onLanguageChange) {
        onLanguageChange(language);
      } else {
        setLocalSelectedLanguage(language);
      }
      setDropdownOpen(false);
      setShowFavoritesOnly(false);
      setLanguageSearchValue('');
      try {
        await updateRecentLanguage({ language });
      } catch (error) {
        console.error('Error updating recent language:', error);
      }
    };
 
    const handleLikeLanguage = async (language, e) => {
      e.stopPropagation();
      if (likingLanguageId === language) return;
      setLikingLanguageId(language);
      const currentFavorites = narratorSettings.favoriteLanguages || [];
      const isCurrentlyFavorite = currentFavorites.some((fav) => fav.language === language);
      if (!isCurrentlyFavorite) {
        setNarratorSettings((prev) => ({
          ...prev,
          favoriteLanguages: [...(prev.favoriteLanguages || []), { language, addedAt: new Date() }],
        }));
      } else {
        setNarratorSettings((prev) => ({
          ...prev,
          favoriteLanguages: (prev.favoriteLanguages || []).filter(
            (fav) => fav.language !== language,
          ),
        }));
      }
      try {
        const result = await toggleFavoriteLanguage({ language }).unwrap();
        if (result?.data) {
          setNarratorSettings((prev) => ({
            ...prev,
            favoriteLanguages: result.data || prev.favoriteLanguages,
          }));
        }
      } catch (error) {
        console.error('Error toggling favorite language:', error);
        setNarratorSettings((prev) => ({
          ...prev,
          favoriteLanguages: currentFavorites,
        }));
      } finally {
        setLikingLanguageId(null);
      }
    };
 
    const toggleDropdown = useCallback(() => {
      setDropdownOpen((wasOpen) => {
        if (wasOpen) {
          setShowFavoritesOnly(false);
          setLanguageSearchValue('');
          return false;
        }
        const el = toggleRef.current;
        if (el) {
          setPortalBox(computePortalLayout(el.getBoundingClientRect()));
        }
        return true;
      });
    }, []);
 
    return (
      <div className={styles.languageDropdown}>
        <p className={styles.languageDropdown__title}>Select language</p>
        <div
          ref={toggleRef}

          >

          </div>
        </div>
        {typeof document !== 'undefined' &&

                top: portalBox.top,
                left: portalBox.left,
                width: portalBox.width,
              }}
              onClick={(e) => e.stopPropagation()}
            >
              <div className={styles.languageDropdown__searchInputContainer}>
                <div className={styles.languageDropdown__searchInputWrapper}>
                  <SearchIcon size="12px" color="rgba(255, 255, 255, 0.32)" />
                  <input
                    type="text"
                    placeholder="Search language"
                    className={styles.languageDropdown__searchInput}
                    value={languageSearchValue}
                    onChange={handleLanguageSearchChange}
                    onKeyDown={handleLanguageSearchKeyDown}
                    maxLength={30}
                  />
                </div>
                <div
                  className={styles.languageDropdown__likeIconWrapper}
                  onClick={(e) => {
                    e.stopPropagation();
                    setShowFavoritesOnly(!showFavoritesOnly);
                  }}
                >
                  <LikeLanguageIcon
                    size="16px"
                    color={showFavoritesOnly ? 'var(--accent-color)' : 'rgba(255, 255, 255, 0.4)'}
                  />
                </div>
              </div>
 
              {showFavoritesOnly && (

                          const language = languages.find(
                            (lang) => lang.value === favoriteLang.language,
                          );
                          if (!language) return null;
                          return { favoriteLang, language };

                        .filter(
                          (item) =>
                            item &&

                              (item.language.name &&
                                item.language.name
                                  .toLowerCase()
                                  .includes(languageSearchValue.toLowerCase())) ||
                              (item.language.value &&
                                item.language.value
                                  .toLowerCase()
                                  .includes(languageSearchValue.toLowerCase()))),

                        .map(({ favoriteLang, language }) => (

                            key={favoriteLang.language}
                            type="button"
                            className={`${styles.languageDropdown__portalItem} ${
                              styles.languageDropdown__portalItemFavorite
                            } ${
                              selectedLanguage === favoriteLang.language
                                ? styles.languageDropdown__portalItemSelected
                                : ''
                            }`}
                            onClick={(e) => {
                              e.stopPropagation();
                              handleLanguageSelect(favoriteLang.language);
                            }}
                          >
                            <div className={styles.languageDropdown__flagContainer}>
                              <Flag
                                country={language.code}
                                size={16}
                                className={styles.languageDropdown__flag}
                              />
                              <div className={styles.languageDropdown__langInfo}>
                                <span>{language.name}</span>
                                {language.description && (
                                  <span className={styles.languageDropdown__langDescription}>
                                    {language.description}
                                  </span>
                                )}
                              </div>
                            </div>
                            <div
                              role="presentation"
                              onClick={(e) => {
                                e.stopPropagation();
                                handleLikeLanguage(favoriteLang.language, e);
                              }}
                            >
                              <LikeLanguageVoiceoverIcon
                                size="14px"
                                color={
                                  likingLanguageId === favoriteLang.language
                                    ? 'rgba(255, 255, 255, 0.2)'
                                    : 'var(--accent-color)'
                                }
                              />
                            </div>
                          </button>
                        ))}
                    </>
                  ) : (
                    <div className={styles.languageDropdown__noFavoritesMessage}>
                      No favorite languages yet
                    </div>
                  )}
                </>
              )}
              {!showFavoritesOnly &&
                languages
                  .filter(
                    (language) =>
                      languageSearchValue === '' ||
                      (language.name &&
                        language.name.toLowerCase().includes(languageSearchValue.toLowerCase())) ||
                      (language.value &&
                        language.value.toLowerCase().includes(languageSearchValue.toLowerCase())),
                  )
                  .map((language) => (
                    <button
                      key={language.value}
                      type="button"
                      className={`${styles.languageDropdown__portalItem} ${
                        selectedLanguage === language.value
                          ? styles.languageDropdown__portalItemSelected
                          : ''
                      }`}
                      onClick={(e) => {
                        e.stopPropagation();
                        handleLanguageSelect(language.value);
                      }}
                    >
                      <div className={styles.languageDropdown__flagContainer}>
                        <Flag
                          country={language.code}
                          size={16}
                          className={styles.languageDropdown__flag}
                        />
                        <div className={styles.languageDropdown__langInfo}>
                          <span>{language.name}</span>
                          {language.description && (
                            <span className={styles.languageDropdown__langDescription}>
                              {language.description}
                            </span>
                          )}
                        </div>
                      </div>
                      <div
                        role="presentation"
                        onClick={(e) => {
                          e.stopPropagation();
                          handleLikeLanguage(language.value, e);
                        }}
                      >
                        <LikeLanguageVoiceoverIcon
                          size="14px"
                          color={
                            likingLanguageId === language.value
                              ? 'rgba(255, 255, 255, 0.2)'
                              : narratorSettings.favoriteLanguages?.some(
                                    (fav) => fav.language === language.value,
                                  )
                                ? 'var(--accent-color)'
                                : 'rgba(255, 255, 255, 0.4)'
                          }
                        />
                      </div>
                    </button>
                  ))}
            </div>,
            document.body,
          )}
      </div>
    );
  },

export function validateSearchInput(value) {
  const allowedChars = /^[a-zA-Z\s\-']*$/;
  return allowedChars.test(value);
}
 
const FontLanguageDropdown = observer(
  ({
    onLanguageChange,
    /** When set, row highlight uses this (e.g. `fontPickerLanguage` from FontSettings). */
    selectedLanguageValue,
    externalLanguageSearch,
    onExternalLanguageSearchChange,
    externalShowFavoritesOnly,
    onExternalShowFavoritesOnlyChange,
  } = {}) => {
  const { data: narratorSettingsData } = useGetNarratorSettingsQuery();
  const [updateRecentLanguage] = useUpdateRecentLanguageMutation();
  const [toggleFavoriteLanguage] = useToggleFavoriteLanguageMutation();
 
  const [selectedLanguage, setSelectedLanguage] = useState('English');
  const [likingLanguageId, setLikingLanguageId] = useState(null);
  const [narratorSettings, setNarratorSettings] = useState({
    recentLanguages: [],
    favoriteLanguages: [],
  });
  const [hoveredLanguage, setHoveredLanguage] = useState(null);
 
  const useExternalLanguageSearch =
    typeof externalLanguageSearch === 'string' &&
    typeof onExternalLanguageSearchChange === 'function';
 
  const useExternalFavorites =
    typeof onExternalShowFavoritesOnlyChange === 'function' &&
    typeof externalShowFavoritesOnly === 'boolean';
 
  const activeSelectedLanguage =
    selectedLanguageValue ?? selectedLanguage;
 
  const effectiveShowFavoritesOnly = useExternalFavorites
    ? externalShowFavoritesOnly
    : false;
 
  const languageSearchForFilter = useExternalLanguageSearch
    ? externalLanguageSearch
    : '';
 
  useEffect(() => {
    if (narratorSettingsData) {
      const settings = narratorSettingsData?.data || {
        recentLanguages: [],
        favoriteLanguages: [],
      };
      setNarratorSettings(settings);
    }
  }, [narratorSettingsData]);
 
  const visibleLanguages = useMemo(() => {
    const favSet = new Set(
      (narratorSettings.favoriteLanguages || []).map((f) => f.language),
    );
    const q = languageSearchForFilter.trim().toLowerCase();
    return FONT_PICKER_LANGUAGES.filter((language) => {
      if (effectiveShowFavoritesOnly && !favSet.has(language.value)) {
        return false;
      }
      if (!q) {
        return true;
      }
      return (
        (language.name && language.name.toLowerCase().includes(q)) ||
        (language.value && language.value.toLowerCase().includes(q))
      );
    });
  }, [
    effectiveShowFavoritesOnly,
    narratorSettings.favoriteLanguages,
    languageSearchForFilter,
  ]);
 
  const emptyLanguagesMessage =
    visibleLanguages.length > 0
      ? null
      : effectiveShowFavoritesOnly
        ? languageSearchForFilter.trim()
          ? 'No favorite languages match your search'
          : 'No favorite languages yet'
        : languageSearchForFilter.trim()
          ? 'No languages match your search'
          : null;
 
  const handleLanguageSelect = async (language) => {
    setSelectedLanguage(language);
    onLanguageChange?.(language);
    if (useExternalFavorites) {
      onExternalShowFavoritesOnlyChange(false);
    }
    if (useExternalLanguageSearch) {
      onExternalLanguageSearchChange('');
    }
    try {
      await updateRecentLanguage({ language });
    } catch (error) {
      console.error('Error updating recent language:', error);
    }
  };
 
  const handleLikeLanguage = async (language, e) => {
    e.stopPropagation();
    if (likingLanguageId === language) return;
    setLikingLanguageId(language);
    const currentFavorites = narratorSettings.favoriteLanguages || [];
    const isCurrentlyFavorite = currentFavorites.some(
      (fav) => fav.language === language,
    );
    if (!isCurrentlyFavorite) {
      setNarratorSettings((prev) => ({
        ...prev,
        favoriteLanguages: [
          ...(prev.favoriteLanguages || []),
          { language, addedAt: new Date() },
        ],
      }));
    } else {
      setNarratorSettings((prev) => ({
        ...prev,
        favoriteLanguages: (prev.favoriteLanguages || []).filter(
          (fav) => fav.language !== language,
        ),
      }));
    }
    try {
      const result = await toggleFavoriteLanguage({ language }).unwrap();
      if (result?.data) {
        setNarratorSettings((prev) => ({
          ...prev,
          favoriteLanguages: result.data || prev.favoriteLanguages,
        }));
      }
    } catch (error) {
      console.error('Error toggling favorite language:', error);
      setNarratorSettings((prev) => ({
        ...prev,
        favoriteLanguages: currentFavorites,
      }));
    } finally {
      setLikingLanguageId(null);
    }
  };
 
  return (
    <div
      className={styles.fontLanguageDropdown}
    >
      <div
        className={styles.fontLanguageDropdown__content}
        onClick={(e) => e.stopPropagation()}
      >
        <div className={styles.fontLanguageDropdown__languages}>
          {emptyLanguagesMessage ? (
            <div className={styles.fontLanguageDropdown__noFavoritesMessage}>
              {emptyLanguagesMessage}
            </div>
          ) : (
            visibleLanguages.map((language) => (
              <button
                key={language.value}
                type="button"
                className={`${styles.fontLanguageDropdown__language} ${activeSelectedLanguage === language.value
                  ? styles.fontLanguageDropdown__contentItemSelected
                  : ''
                  }`}
                onClick={(e) => {
                  e.stopPropagation();
                  handleLanguageSelect(language.value);
                }}
              >
                <div className={styles.fontLanguageDropdown__flagContainer}>
                  <Flag
                    country={language.code}
                    size={12}
                    className={styles.languageDropdown__flag}
                  />
                  <span>{language.name}</span>
                </div>
                <div
                  role="button"
                  tabIndex={0}
                  className={styles.fontDropdown_itemLike}
                  onClick={e => {
                    e.stopPropagation();
                    handleLikeLanguage(language.value, e);
                  }}
                  onKeyDown={e => {
                    if (e.key === 'Enter' || e.key === ' ') {
                      e.preventDefault();
                      e.stopPropagation();
                      handleLikeLanguage(language.value, e);
                    }
                  }}
                  onMouseEnter={() =>
                    setHoveredLanguage(language.value)
                  }
                  onMouseLeave={() => setHoveredLanguage(null)}
                >
                  <LikeLanguageVoiceoverIcon
                    size="14px"
                    opacity={1}
                    color={
                      narratorSettings.favoriteLanguages?.some(
                        (fav) => fav.language === language.value,
                      )
                        ? 'var(--accent-color)'
                        : hoveredLanguage === language.value
                          ? '#FFF'
                          : 'rgba(255, 255, 255, 0.4)'
                    }
                  />
                </div>
              </button>
            ))
          )}
        </div>
      </div>
    </div>
  );
});
 
export default FontLanguageDropdown;

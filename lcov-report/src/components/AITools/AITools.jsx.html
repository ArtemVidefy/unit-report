// Import tool only for QA and Admin users
const importTool = {
  id: 5,
  name: 'IMPORT',
  title: 'Import Project',
  description: 'Import a project from\nJSON file',
  icon: 'ImportIcon',
  color: '#FFB84D',
  isImport: true,
};
 
const IconMap = Object.fromEntries(
  Object.keys(Icons).map(key => [key, Icons[key]])
);
 
const AITools = ({
  handleToolClick,
  hasBorder = false,
  toolsWidth = '392px',
  disableModal = false,
  flexDirection = 'row',
  onImportProject,
}) => {
  const scrollContainerRef = useHorizontalScroll();
  const [activeTool, setActiveTool] = useState(null);
  const [isCreateVideoModalOpen, setIsCreateVideoModalOpen] = useState(false);
  const navigate = useNavigate();
  const { username, type: userType } = useSelector(user);
  const [addStory] = useAddStoryMutation();
 
  // Show import tool only for QA and Admin
  const isQAorAdmin = userType === 'QA' || userType === 'Admin';
  const availableTools = isQAorAdmin ? [...toolsList, importTool] : toolsList;
 
  const handleClick = tool => {
    setActiveTool(tool.name);
    handleToolClick(tool.name);
 
    if (!tool.disabled) {
      trackEvent(`${tool.title} clicked`, { page_name: "home" });
      if (tool.name === 'MYSCRIPT') {
        trackVideoCreation({
          location: AnalyticsSurfaceLocation.HOME,
          button_name: VideoCreationButtonNames.START_CREATE_A_VIDEO,
        });
      }
      if (tool.name === 'MYSCRIPT' && !disableModal) {
        setIsCreateVideoModalOpen(true);
      } else if (tool.name === 'MYSCRIPT' && disableModal) {
        navigate(tool.path);
      } else if (tool.isBlank) {
        // For blank projects, handle creation logic here
        handleBlankProjectCreation();
      } else if (tool.isImport) {
        // Handle import
        handleImportProject();
      } else if (tool.path) {
        navigate(tool.path);
      }
    }
  };
 
  const handleBlankProjectCreation = async () => {
    // Create blank project and redirect to VideoCreationPage
    try {
      const defaultSize = sizes.sizes[0]; // Default size
      const defaultStyle = sizes.presets[1]; // Default style
 
      const defaultMaxTime = 60000; // 60 seconds default
 
      const newStory = await addStory({
        name: '',
        text: '',
        isDraft: false,
        author: username,
        width: defaultSize.size.width,
        height: defaultSize.size.height,
        orientation: defaultSize.name,
        generationStyle: {
          preset: defaultStyle?.preset || 'dynamic',
          style: defaultStyle?.style,
          modelId: defaultStyle?.modelId,
        },
        resolution: defaultSize.generationSize,
        skipVoiceover: true,
        maxTime: defaultMaxTime,
        characters: [],
        isBlank: true,
      }).unwrap();
 
      // Redirect to VideoCreationPage with blank flag
      navigate(`/createVideo/${newStory._id}?blank=true`);
    } catch (error) {
      console.error('Error creating blank project:', error);
      console.error('Error details:', error.response?.data || error.message);
      // Show error to user
      alert('Failed to create blank project. Please try again.');
    }
  };
 
  const handleImportProject = () => {
    // Create a hidden file input
    const fileInput = document.createElement('input');
    fileInput.type = 'file';
    fileInput.accept = '.json';
    fileInput.style.display = 'none';
 
    fileInput.onchange = async (e) => {
      const file = e.target.files[0];
      if (!file) return;
 
      try {
        const text = await file.text();
        const storyData = JSON.parse(text);
 
        // Call the import handler passed from parent
        if (onImportProject) {
          await onImportProject(storyData);
        }
      } catch (error) {
        console.error('Error importing project:', error);
        alert('Failed to import project. Please check the file format.');
      } finally {
        document.body.removeChild(fileInput);
      }
    };
 
    document.body.appendChild(fileInput);
    fileInput.click();
  };
 
  return (
    <>
      <div className={styles.container} style={{ flexDirection }} ref={scrollContainerRef} >
        {availableTools.map(tool => {
          const IconComponent = IconMap[tool.icon];
          return (
            <div
              key={tool.id}
              className={`${styles.tool} ${
                activeTool === tool.name ? styles.active : ''
              } ${tool.disabled ? styles.disabled : ''} ${
                hasBorder ? styles.bordered : ''
              }`}
              style={{ width: toolsWidth, flexShrink: 0 }}
              onClick={() => !tool.disabled && handleClick(tool)}
            >
              {/* <img src={tool.icon} alt={tool.name} className={styles.icon} /> */}
              <div className={styles.icon}>
                <EffectIcon color={tool.color} />
                <div className={styles.iconContent}>
                  <IconComponent color={tool.color} />
                </div>
              </div>
              <div className={styles.toolContent}>
                <h3 className={styles.title}>{tool.title}</h3>
                <p className={styles.description}>{tool.description}</p>
              </div>
              {tool.new && (
                <p className={styles.comingSoon}>New</p>
              )}
              {tool.comingSoon && (
                <p className={styles.comingSoon}>coming soon</p>
              )}
            </div>
          );
        })}
      </div>
      {isCreateVideoModalOpen && (
        <CreateVideoModal closeModal={() => setIsCreateVideoModalOpen(false)} />
      )}
    </>
  );
};
 
export { AITools };

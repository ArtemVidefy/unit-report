import { Fragment, useState, useEffect } from 'react';
import styles from './GenerateWithAiSection.module.scss';
import { useHorizontalScroll } from '../../hooks/useHorizontalScroll';
import { Tooltip } from 'react-tooltip';
import InfoIconSmall from 'components/Icons/InfoIconSmall';
import TransitionsTooltipIcon from 'components/Icons/TransitionsTooltipIcon';
import EffectsTooltipIcon from 'components/Icons/EffectsTooltipIcon';
import OverlaysTooltipIcon from 'components/Icons/OverlaysTooltipIcon';
import FiltersTooltipIcon from 'components/Icons/FiltersTooltipIcon';
import { ButtonWithIcon } from 'components/reusableComponents/ButtonWithIcon';
 
const GenerateAIModeComponent = ({
  activeTab,
  handleTabClick,
  tabs,
  containerStyle,
  tabsStyle,
  tabStyle,
  activeTabStyle,
  disabledTabStyle,
  isLeftPanel = false,
  containerWidth: propContainerWidth,
}) => {
  const scrollContainerRef = useHorizontalScroll();
  const [internalContainerWidth, setInternalContainerWidth] = useState(0);
 
  useEffect(() => {
    if (!scrollContainerRef.current || propContainerWidth) return;
 
    const updateWidth = () => {
      if (scrollContainerRef.current) {
        const width = scrollContainerRef.current.offsetWidth;
        setInternalContainerWidth(width);
      }
    };
 
    updateWidth();
 
    const resizeObserver = new ResizeObserver(() => {
      updateWidth();
    });
 
    resizeObserver.observe(scrollContainerRef.current);
 
    return () => {
      resizeObserver.disconnect();
    };
  }, [scrollContainerRef, propContainerWidth]);
 
  const containerWidth = propContainerWidth || internalContainerWidth;
 
  // Calculate responsive font size and padding based on available width per tab
  const getResponsiveStyles = () => {
    const defaultFontSize = 12;
    const defaultPadding = '3px 10px';
 
    const shouldReduceSize = isLeftPanel || (containerWidth > 0 && containerWidth < 300);
 
    if (!shouldReduceSize || !containerWidth || tabs.length === 0) {
      return { fontSize: defaultFontSize, padding: defaultPadding };
    }
 
    const dividerWidth = 1;
    const dividersCount = tabs.length - 1;
    const borderWidth = 2;
    const availableWidth = containerWidth - dividersCount * dividerWidth - borderWidth;
    const widthPerTab = availableWidth / tabs.length;
 
    const minWidthPerTab = 20;
    const maxWidthPerTab = 100;
    const minFontSize = 12;
    const maxFontSize = 12;
    const minPaddingX = 4;
    const maxPaddingX = 10;
    const minPaddingY = 2;
    const maxPaddingY = 3;
 
    if (widthPerTab >= maxWidthPerTab) {
      return {
        fontSize: maxFontSize,
        padding: `${maxPaddingY}px ${maxPaddingX}px ${maxPaddingY + 2}px ${maxPaddingX}px`,
      };
    }
 
    if (widthPerTab <= minWidthPerTab) {
      return {
        fontSize: minFontSize,
        padding: `${minPaddingY}px ${minPaddingX}px ${minPaddingY + 2}px ${minPaddingX}px`,
      };
    }
 
    const ratio = (widthPerTab - minWidthPerTab) / (maxWidthPerTab - minWidthPerTab);
    const fontSize = Math.round(minFontSize + (maxFontSize - minFontSize) * ratio);
    const paddingX = Math.round(minPaddingX + (maxPaddingX - minPaddingX) * ratio);
    const paddingY = Math.round(minPaddingY + (maxPaddingY - minPaddingY) * ratio);
 
    return {
      fontSize,
      padding: `${paddingY}px ${paddingX}px ${paddingY + 2}px ${paddingX}px`,
    };
  };
 
  const responsiveStyles = getResponsiveStyles();
  const tooltipContentByTab = {
    Transitions: {
      icon: <TransitionsTooltipIcon />,
      title: 'Seamlessly connect your scenes',
      description: 'Use transitions to guide viewers between moments with flow and intention.',
    },
    Effects: {
      icon: <EffectsTooltipIcon />,
      title: 'Add visual impact and energy',
      description: 'Enhance your scenes with motion, glow, particles, or subtle visual depth.',
    },
    Overlays: {
      icon: <OverlaysTooltipIcon />,
      title: 'Layer atmosphere and texture',
      description: 'Overlays let you build mood, tone, or motion without changing the base footage.',
    },
    Filters: {
      icon: <FiltersTooltipIcon />,
      title: 'Set the tone in one click',
      description: 'Change the look and feel of your video with curated color and lighting styles.',
    },
  };
  const activeTooltip = tooltipContentByTab[activeTab];
  const activeTooltipId = 'visual-fx-active-tab-tooltip';
 
  return (
    <div className={styles.tabContainer} style={containerStyle} ref={scrollContainerRef}>
      <div className={styles.tabsWithInfo}>
        <div className={styles.tabs} style={tabsStyle}>
          {tabs.map((tab, idx) => (
            <Fragment key={tab.label}>
              <button
                onClick={() => handleTabClick(tab.label)}
                className={`${styles.tab} ${activeTab === tab.label ? styles.active : ''}`}
                disabled={!tab.active}
                style={{
                  fontSize: `${responsiveStyles.fontSize}px`,
                  padding: responsiveStyles.padding,
                  ...tabStyle,
                  ...(activeTab === tab.label && activeTabStyle),
                  ...(!tab.active && disabledTabStyle),
                }}
              >
                <span className={styles.tabLabel}>{tab.label}</span>
              </button>
              {idx < tabs.length - 1 && (
                <span
                  className={`${styles.tabDivider} ${
                    tab.label !== activeTab && tabs[idx + 1].label !== activeTab
                      ? styles.visible
                      : styles.hidden
                  }`}
                ></span>
              )}
            </Fragment>
          ))}
        </div>
        {activeTooltip && (
          <>
            <button
              className={styles.tabInfoIcon}
              data-tooltip-id={activeTooltipId}
              data-tooltip-place="bottom"
              onClick={(e) => {
                e.preventDefault();
                e.stopPropagation();
              }}
              aria-label={`Show ${activeTab} tooltip`}
              type="button"
            >
              <ButtonWithIcon icon='QuestionCircleIcon' size="10px" color="rgba(255,255,255,0.5)" accentColor='white' activeColor='var(--accent-color)'/>
            </button>
            <Tooltip
              id={activeTooltipId}
              className={styles.visualFxTooltip}
              classNameArrow={styles.visualFxTooltipArrow}
              delayShow={150}
              offset={8}
              clickable
              place="bottom"
            >
              <div className={styles.tooltipContent}>
                <div className={styles.visualFxTooltipHeader}>
                  <span className={styles.visualFxTooltipIcon}>{activeTooltip.icon}</span>
                  <span className={styles.visualFxTooltipTitle}>{activeTooltip.title}</span>
                </div>
                <p className={styles.visualFxTooltipDescription}>{activeTooltip.description}</p>
              </div>
            </Tooltip>
          </>
        )}
      </div>
    </div>
  );
};
 
export { GenerateAIModeComponent };

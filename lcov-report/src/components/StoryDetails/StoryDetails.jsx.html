import styles from './StoryDetails.module.scss';
import { ButtonWithIcon } from 'components/reusableComponents/ButtonWithIcon';
import { useEffect, useState } from 'react';
 
const StoryDetails = ({
  isEmptyStoryName,
  isEmptyStory,
  currentStoryContent = '',
  currentStoryName = '',
  onUpdateStory,
  customPadding,
  storyTitle,
  textTitle,
}) => {
  const [storyName, setStoryName] = useState(currentStoryName);
  const [storyContent, setStoryContent] = useState(currentStoryContent);
  const [storyNameTouched, setStoryNameTouched] = useState(false);
  const [storyContentTouched, setStoryContentTouched] = useState(false);
  const [nameError, setNameError] = useState('');
  const [contentError, setContentError] = useState('');
 
  const handleStoryNameChange = e => {
    const newName = e.target.value;
    if (newName.length <= 50) {
      setStoryName(newName);
      setNameError('');
      onUpdateStory(newName, storyContent);
      if (!storyNameTouched) {
        setStoryNameTouched(true);
      }
 
      if (newName.length === 50) {
        setNameError("You've reached the maximum length.");
      } else if (newName.length >= 3 && newName.length < 50) {
        setNameError('');
      }
    }
  };
 
  const handleStoryContentChange = e => {
    let newContent = e.target.value;
 
    // Capitalize first letter of the text
    newContent = newContent.replace(/^([a-zа-яґєії])/i, letter =>
      letter.toUpperCase()
    );
 
    // Capitalize letters after periods
    newContent = newContent.replace(/\.\s*([a-zа-яґєії])/gi, match => {
      return match.replace(/[a-zа-яґєії]/i, letter => letter.toUpperCase());
    });
 
    setStoryContent(newContent);
    onUpdateStory(storyName, newContent);
 
    if (!storyContentTouched) {
      setStoryContentTouched(true);
    }
    if (newContent.length < 8) {
      setContentError('Story should be at least 8 symbols.');
    } else {
      setContentError('');
    }
  };
 
  useEffect(() => {
    setStoryName(currentStoryName);
    setStoryContent(currentStoryContent);
  }, [currentStoryName, currentStoryContent]);
 
  useEffect(() => {
    if (storyNameTouched) {
      const timer = setTimeout(() => {
        if (storyName.length < 3) {
          setNameError('Please provide at least 3 characters.');
        } else if (storyName.length === 50) {
          setNameError("You've reached the maximum length.");
        } else {
          setNameError('');
        }
      }, 500);
 
      return () => clearTimeout(timer);
    }
  }, [storyName, storyNameTouched]);
 
  useEffect(() => {
    if (storyContentTouched) {
      const timer = setTimeout(() => {
        if (storyContent.length < 8) {
          setContentError('Please provide at least 8 characters.');
        } else {
          setContentError('');
        }
      }, 500);
 
      return () => clearTimeout(timer);
    }
  }, [storyContent, storyContentTouched]);
 
  return (
    <form
      className={styles.creation_inputForm}
      onSubmit={e => e.preventDefault()}
      style={customPadding}
    >
      <div className={styles.input_box}>
        <label>{storyTitle}</label>
        <div className={styles.inputWrapper}>
          <input
            placeholder={
              isEmptyStoryName ? 'This field can not be empty!' : 'Type here'
            }
            className={`${styles.creation_input} ${
              nameError || isEmptyStoryName ? styles.input_error : ''
            }`}
            value={storyName}
            onChange={handleStoryNameChange}
            maxLength={51}
          />
          <ButtonWithIcon
            icon="ChatGptIcon"
            tooltipText="Generate story name"
            classNameButton={styles.icon}
          />
          {storyNameTouched && nameError && (
            <span className={styles.error_message}>{nameError}</span>
          )}
        </div>
      </div>
      <div className={styles.input_box}>
        <label>{textTitle}</label>
        <textarea
          placeholder={
            isEmptyStory ? 'This field can not be empty!' : 'Type here'
          }
          className={`${styles.creation_textArea} ${
            contentError ? styles.input_error : ''
          }`}
          value={storyContent}
          onChange={handleStoryContentChange}
        ></textarea>
        {storyContentTouched && contentError && (
          <span className={styles.error_message}>{contentError}</span>
        )}
      </div>
    </form>
  );
};
 
export { StoryDetails };

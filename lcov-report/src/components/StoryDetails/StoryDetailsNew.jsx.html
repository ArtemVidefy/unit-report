                value={storyContent}
                onChange={isSettingsPage ? undefined : handleStoryContentChange}
                onBlur={isSettingsPage ? undefined : handleContentBlur}
                placeholder={
                  isEmptyStory ? 'This field can not be empty!' : 'Type your story here...'
                }
                maxLength={maxStoryContentLength}
                showCopyButton={true}
                showStats={true}
                stats={localStats}
                showDuration={selectedScriptMode !== 'idea'}
                className={contentError ? styles.input_error : ''}
                isRefining={isRefining}
                disabled={isRefining}
                readOnly={isSettingsPage}
                showSceneWatermark={selectedScriptMode === 'scene-breakdown' && !isSettingsPage}
                onAddScene={
                  selectedScriptMode === 'scene-breakdown' && !isSettingsPage
                    ? handleAddScene
                    : undefined
                }
              />
              {storyContentTouched && contentError && (
                <span className={styles.error_message}>{contentError}</span>
              )}
            </>
          )}
          {!hideStoryPrompt && !isSettingsPage && (
            <div className={styles.actionButtons}>
              {/* Changes detected notification */}
              {showRefineChangesNotification && (
                <div className={styles.refine_changes_notification}>
                  <span className={styles.refine_changes_message}>
                    Changes detected. Would you like to start over?
                  </span>
                  <div className={styles.refine_changes_buttons}>
                    <button className={styles.refine_changes_button} onClick={onRefineChangesNo}>
                      No
                    </button>
                    <button
                      className={`${styles.refine_changes_button} ${styles.refine_changes_button_yes}`}
                      onClick={onRefineChangesYes}
                    >
                      Yes
                    </button>
                  </div>
                </div>
              )}
              <div className={styles.actionButtonsRow}>
                <ButtonWithIcon
                  text={isRefining ? 'Refining...' : getRefineButtonText()}
                  onClick={handleRefineClick}
                  classNameButton={styles.refineButton}
                  disabled={arePrimaryActionsDisabled || isRefining}
                  marginLeft="0"
                >
                  {isRefining && (
                    <span
                      style={{ marginLeft: 8, display: 'inline-flex', verticalAlign: 'middle' }}
                    >
                      <LoaderDots />
                    </span>
                  )}
                </ButtonWithIcon>
                <Button
                  text="Continue"
                  handleClick={handleContinueClick}
                  // className={styles.continueButton}
                  isDisabled={
                    arePrimaryActionsDisabled ||
                    isGeneratingIdea ||
                    isGeneratingNameFromContinue ||
                    isContinueClicked
                  }
                  variant="contained"
                  title={continueTooltip}
                />
              </div>
            </div>
          )}
        </div>
      </form>
    );
  },
);

    // Let react-dnd/internal handlers process non-file drags (gallery items, scene reordering, etc.)
    if (!externalFileDrop.isExternalFileDrag(e)) {
      return;
    }
 
    // Stop propagation so external file drops don't also trigger other handlers.
    e.stopPropagation();
    setIsUploading(true);
    const result = await handleExternalFileDrop(
      e,
      externalFileDrop.isExternalFileDrag,
      scene?._id,
      store,
      storyData,
      dispatch,
      updateSelectedImage,
      externalFileDrop.createdBlobUrlsRef,
      externalFileDrop.clearDragState,
      (val) => {},
      (val) => {},
      { current: 0 },
      storageUsageAvailable
    );
    setIsUploading(false);
    if (result?.error) {
      showError(result.error);
    } else if (result?.success) {
      dispatch(fetchSubscriptionUsage());
      if (onFileDropped) onFileDropped();
    }
  };
 
  const externalDropHoverClass = (() => {
    if (externalFileDrop.isExternalDragging) {
      if (externalFileDrop.isExternalValidDrag === false) {
        return styles.invalidDropHover;
      }
      return styles.imageDropHover;
    }
    if (isOverInternalGallery) return styles.imageDropHover;
    return '';
  })();
 
  return (
    <div
      ref={dropRef}
      className={`${className || ''} ${externalDropHoverClass}`}
      onDragEnter={externalFileDrop.handleExternalDragEnter}
      onDragLeave={externalFileDrop.handleExternalDragLeave}
      onDragOver={(e) => externalFileDrop.throttledDragOverRef.current?.(e)}
      onDrop={handleDrop}
      data-scene-dropzone="true"
      style={{ position: 'relative' }}
    >
      {children}
    </div>
  );
});

import React, {
  useState,
  useRef,
  useEffect,
  useLayoutEffect,
  useMemo,
  useId,
  useCallback,
} from 'react';
import { useSelector } from 'react-redux';
import { user as selectUser } from '../../redux/auth/selectors';
import styles from './GenerationModeToggle.module.scss';
import { ButtonWithIcon } from 'components/reusableComponents/ButtonWithIcon';
import { motion, LayoutGroup, AnimatePresence } from 'framer-motion';
import GenModeToggleSegment from './GenModeToggleSegment/GenModeToggleSegment';
import { DEFAULT_GEN_MODE_TOGGLE_ITEMS, TAB_MODES } from './constants';
import { buildPrioritizedLayoutSegments } from './utils/buildPrioritizedLayoutSegments';
import { getItemPosition } from './utils/genModeToggleUtils';
import GenModeToggleModal from './GenModeToggleModal/GenModeToggleModal';
import GenModeDropdown from './GenModeDropdown/GenModeDropdown';
import { getGenModeToggleDividerVisibility } from './utils/getGenModeToggleDividerVisibility';
import {
  GEN_MODE_TOGGLE_TABS_MODE_CHANGED_EVENT,
  getInitialPersistedTabsMode,
  loadGenModeToggleTabsMode,
  saveGenModeToggleTabsMode,
} from './utils/genModeToggleUiPersistence';
 
const OVERFLOW_MODE_IDS = ['audioToVideo', 'videoToVideo', 'avatar', 'upscale'];
 
const VIDEO_MODES = [
  'textToVideo',
  'imageToVideo',
  'videoToVideo',
  'audioToVideo',
  'avatar',
  'upscale',
];
 
const IMAGE_MODES = [
  'textToImage',
  'textToVideo',
  'imageToImage',
  'imageToVideo',
  'avatar',
  'upscale',
];
 
const PRIMARY_COLLAPSED_COUNT = 4;
 
// Layout transition duration default duration
const GEN_MODE_TOGGLE_LAYOUT_DURATION = 0.3;
 
const GenerationModeToggle = ({
  activeMode = 'textToImage',
  onModeChange,
  disabled = false,
  allowedModes,
  fullSizeModesPreset = null,
  onHoverModeChange,
  defaultTabsMode = 'expanded',
  compactToolbar = false,
  persistenceScope = null, // When set, toolbar layout is loaded/saved per scope in localStorage.
  expandAllTabsOnFirstItemHover = true, //When true (default), hovering the first tab sets shared `isFirstHovered` so peer tabs expand too.
}) => {
  const user = useSelector(selectUser);
  const isBetaUser = Boolean(user?.betaUser);
 
  const [isGenModeToggleModalOpen, setIsGenModeToggleModalOpen] = useState(false);
  const [isDropdownOpen, setIsDropdownOpen] = useState(false);
  const [hoveredItem, setHoveredItem] = useState(null);
  const [isFirstHovered, setIsFirstHovered] = useState(false);
 
  const genModeToggleButtonRef = useRef(null);
  const itemRefs = useRef({});
 
  const genModeToggleItems = useMemo(() => {
    return DEFAULT_GEN_MODE_TOGGLE_ITEMS.filter((item) => {
      if (isBetaUser && item.id === 'avatar') {
        return false;
      }
      if (allowedModes && Array.isArray(allowedModes)) {
        return allowedModes.includes(item.id);
      }
 
      if (fullSizeModesPreset === 'videoAsset') {
        return VIDEO_MODES.includes(item.id);
      }
      if (fullSizeModesPreset === 'imageAsset') {
        return IMAGE_MODES.includes(item.id);
      }
 
      return true;
    });
  }, [isBetaUser, allowedModes, fullSizeModesPreset]);
 
  const activeTitleFromMode = useMemo(() => {
    const match = DEFAULT_GEN_MODE_TOGGLE_ITEMS.find((item) => item.id === activeMode);
    return match?.title ?? DEFAULT_GEN_MODE_TOGGLE_ITEMS[0]?.title;
  }, [activeMode]);
 
  const genModeToggleLayoutTransition = {
    layout: {
      duration: GEN_MODE_TOGGLE_LAYOUT_DURATION,
      ease: [0.22, 1, 0.36, 1],
    },
  };
  const [activeItem, setActiveItem] = useState(activeTitleFromMode);
 
  useEffect(() => {
    setActiveItem(activeTitleFromMode);
  }, [activeTitleFromMode]);
 
  const selectNavItem = useCallback(
    (title, { skipDropdownDeselect } = {}) => {
      const item = genModeToggleItems.find((n) => n.title === title);
      if (!item || disabled) {
        return;
      }
      if (!skipDropdownDeselect && OVERFLOW_MODE_IDS.includes(item.id) && item.id === activeMode) {
        if (onModeChange) {
          onModeChange('textToImage');
        }
        const textToImageTitle = genModeToggleItems.find((n) => n.id === 'textToImage')?.title;
        setActiveItem(textToImageTitle ?? genModeToggleItems[0]?.title);
        return;
      }
      if (onModeChange) {
        onModeChange(item.id);
      }
      setActiveItem(title);
    },
    [genModeToggleItems, disabled, onModeChange, activeMode],
  );
  const [tabsMode, setTabsMode] = useState(() =>
    getInitialPersistedTabsMode(persistenceScope, defaultTabsMode),
  );
  const skipNextTabsModePersist = useRef(true);
 
  useLayoutEffect(() => {
    skipNextTabsModePersist.current = true;
    if (!persistenceScope) return;
    const saved = loadGenModeToggleTabsMode(persistenceScope);
    if (saved) {
      setTabsMode(saved);
    }
  }, [persistenceScope]);
 
  useEffect(() => {
    if (!persistenceScope) return;
    if (skipNextTabsModePersist.current) {
      skipNextTabsModePersist.current = false;
      return;
    }
    saveGenModeToggleTabsMode(persistenceScope, tabsMode);
  }, [persistenceScope, tabsMode]);
 
  // Since we can have multiple toggles on the same page, we need to sync the tabs layout when another instance saves the same scope.
  // So we emit custom event when another instance saves the same scope to trigger a re-render of the tabs layout.
  useEffect(() => {
    if (!persistenceScope || typeof window === 'undefined') {
      return undefined;
    }
    const allowedIds = new Set(TAB_MODES.map((m) => m.id));
    const onSameScopeTabsModeSaved = (e) => {
      const { scope, tabsMode: next } = e.detail || {};
      if (scope !== persistenceScope || typeof next !== 'string' || !allowedIds.has(next)) return;
 
      setTabsMode((prev) => (prev === next ? prev : next));
    };
    window.addEventListener(GEN_MODE_TOGGLE_TABS_MODE_CHANGED_EVENT, onSameScopeTabsModeSaved);
    return () => {
      window.removeEventListener(GEN_MODE_TOGGLE_TABS_MODE_CHANGED_EVENT, onSameScopeTabsModeSaved);
    };
  }, [persistenceScope]);
 
  //Unique layoutId prefix for the instance so framer motion doesn't touch multiple toolbars on the page at once
  const layoutIdPrefix = useId().replace(/:/g, '');
 
  const dropdownLinks = useMemo(() => {
    return genModeToggleItems.slice(PRIMARY_COLLAPSED_COUNT);
  }, [genModeToggleItems]);
 
  const dropdownTitleSet = new Set(dropdownLinks.map((item) => item.title));
  const isPrioritizedModeLayout = tabsMode === 'collapsed' || tabsMode === 'priorityTabs';
  const showCollapsedOverflowSlot = isPrioritizedModeLayout && dropdownTitleSet.has(activeItem);
  const setItemRef = (title, element) => {
    itemRefs.current[title] = element;
  };
 
  const renderNavSegment = (
    item,
    collapsedShowLabel,
    hasDividerSlot,
    isDividerVisible,
    itemPosition = 'single',
    isTemporaryItemShown = false,
  ) => (
    <GenModeToggleSegment
      key={item.title}
      item={item}
      collapsedShowLabel={collapsedShowLabel}
      hasDividerSlot={hasDividerSlot}
      isDividerVisible={isDividerVisible}
      itemPosition={itemPosition}
      isTemporaryItemShown={isTemporaryItemShown}
      setItemRef={setItemRef}
      activeItem={activeItem}
      setActiveItem={selectNavItem}
      tabsMode={tabsMode}
      hoveredItem={hoveredItem}
      setHoveredItem={setHoveredItem}
      isFirstHovered={isFirstHovered}
      setIsFirstHovered={setIsFirstHovered}
      firstItemTitle={DEFAULT_GEN_MODE_TOGGLE_ITEMS[0].title}
      layoutTransition={genModeToggleLayoutTransition}
      layoutIdPrefix={layoutIdPrefix}
      onHoverModeChange={onHoverModeChange}
      expandAllTabsOnFirstItemHover={expandAllTabsOnFirstItemHover}
    />
  );
 
  return (
    <div
      className={`${styles.genModeToggleContainer} ${compactToolbar ? styles.genModeToggleContainerCompact : ''}`}
    >
      <LayoutGroup>
        <nav className={styles.genModeToggle}>
          {isPrioritizedModeLayout ? (
            <AnimatePresence initial={false}>
              {(() => {
                const segments = buildPrioritizedLayoutSegments({
                  genModeToggleItems,
                  activeItem,
                  tabsMode,
                  dropdownLinks,
                  showCollapsedOverflowSlot,
                  hideOverflowDivider: isDropdownOpen || dropdownTitleSet.has(activeItem),
                  renderNavSegment,
                });
                if (dropdownLinks.length > 0) {
                  segments.push(
                    <motion.div
                      key="__overflow"
                      layout={showCollapsedOverflowSlot ? false : 'position'}
                      className={styles.genModeToggleItemWithDivider}
                      transition={genModeToggleLayoutTransition}
                    >
                      <GenModeDropdown
                        dropdownItems={dropdownLinks}
                        activeItem={activeItem}
                        setActiveItem={selectNavItem}
                        onOpenChange={setIsDropdownOpen}
                        toolbarHoveredItem={hoveredItem}
                        toolbarIsFirstHovered={isFirstHovered}
                      />
                    </motion.div>,
                  );
                }
 
                return segments;
              })()}
            </AnimatePresence>
          ) : (
            genModeToggleItems.map((item, index) => {
              const itemPosition = getItemPosition(index, genModeToggleItems.length);
              const nextItem = genModeToggleItems[index + 1];
              const nextItemPosition = nextItem
                ? getItemPosition(index + 1, genModeToggleItems.length)
                : 'single';
              const hasTrailingItem = index < genModeToggleItems.length - 1;
              const isDividerVisible = getGenModeToggleDividerVisibility({
                activeItem,
                item,
                itemPosition,
                nextItem,
                nextItemPosition,
              });
 
              return renderNavSegment(item, false, hasTrailingItem, isDividerVisible, itemPosition);
            })
          )}
        </nav>
 
        {!compactToolbar && (
          <ButtonWithIcon
            ref={genModeToggleButtonRef}
            icon="GenerationModeToggleModalIcon"
            opacity="1"
            size="16px"
            color="rgba(255, 255, 255, 0.4)"
            accentColor="rgba(255, 255, 255, 1)"
            activeColor="rgba(211, 248, 90, 1)"
            onClick={() => {
              setIsGenModeToggleModalOpen(!isGenModeToggleModalOpen);
            }}
            classNameButton={
              isGenModeToggleModalOpen
                ? `${styles.genModeToggleExpandButton} ${styles.activeGenModeToggleExpandButton}`
                : styles.genModeToggleExpandButton
            }
          />
        )}
        <GenModeToggleModal
          anchorRef={genModeToggleButtonRef}
          open={isGenModeToggleModalOpen}
          setIsGenModeToggleModalOpen={setIsGenModeToggleModalOpen}
          setTabsMode={setTabsMode}
          tabsMode={tabsMode}
          layoutIdPrefix={layoutIdPrefix}
          previewActiveItemTitle={activeItem}
        />
      </LayoutGroup>
    </div>
  );
};
 
export { GenerationModeToggle };

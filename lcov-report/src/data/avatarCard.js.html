export const getAvatarByGenderAndVoiceId = (gender, voiceId) => {
  if (!voiceId) {
    if (gender === 'male') {
      return maleAvatars[0];
    } else if (gender === 'female') {
      return femaleAvatars[0];
    }
    return femaleAvatars[0];
  }
  
  let hash = 0;
  for (let i = 0; i < voiceId.length; i++) {
    const char = voiceId.charCodeAt(i);
    hash = ((hash << 5) - hash) + char;
    hash = hash & hash; 
  }
  
  if (gender === 'male') {
    const index = Math.abs(hash) % maleAvatars.length;
    return maleAvatars[index];
  } else if (gender === 'female') {
    const index = Math.abs(hash) % femaleAvatars.length;
    return femaleAvatars[index];
  }
  
  const index = Math.abs(hash) % femaleAvatars.length;
  return femaleAvatars[index];
};

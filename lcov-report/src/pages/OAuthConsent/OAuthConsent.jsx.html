import { useEffect, useMemo, useState } from 'react';
import { useSearchParams } from 'react-router-dom';
import axios from 'axios';
import { useSelector } from 'react-redux';
import './oauthConsent.scss';
 
/**
 * OAuth 2.1 consent screen rendered after Claude (or any other OAuth client)
 * redirects the user to /oauth/consent with the request parameters echoed
 * from /oauth/authorize on the backend.
 *
 * The user must already be signed in via Clerk — the App router gates this
 * page behind <SignedIn>. We POST the user's decision to
 * /oauth/authorize/decision (Clerk-protected) and then navigate the browser
 * to the callback URL returned by the backend.
 */
export const OAuthConsent = () => {
  const [params] = useSearchParams();
  const token = useSelector(state => state.auth?.token);
 
  const [busy, setBusy] = useState(false);
  const [error, setError] = useState(null);
 
  const clientName = params.get('client_name') || 'MCP Client';
  const clientId = params.get('client_id');
  const redirectUri = params.get('redirect_uri');
  const scope = params.get('scope') || 'videfy.read videfy.write';
 
  const scopes = useMemo(
    () => scope.split(/\s+/).filter(Boolean),
    [scope],
  );
 
  const missingFields = !clientId || !redirectUri || !params.get('code_challenge');
 
  useEffect(() => {
    document.title = `Authorize ${clientName} · Videfy`;
  }, [clientName]);
 
  const submit = async (decision) => {
    if (busy || missingFields) return;
    setBusy(true);
    setError(null);
    try {
      const body = {
        decision,
        client_id: clientId,
        redirect_uri: redirectUri,
        code_challenge: params.get('code_challenge'),
        code_challenge_method: params.get('code_challenge_method') || 'S256',
        state: params.get('state') || undefined,
        scope,
        resource: params.get('resource') || undefined,
      };
      const { data } = await axios.post(
        `${process.env.REACT_APP_BACKEND_URL}oauth/authorize/decision`,
        body,
        { headers: { Authorization: `Bearer ${token}` } },
      );
      if (data?.redirectTo) {
        window.location.href = data.redirectTo;
      } else {
        setError('Backend did not return a redirect URL.');
        setBusy(false);
      }
    } catch (err) {
      setError(
        err?.response?.data?.error_description ||
          err?.response?.data?.error ||
          err?.message ||
          'Failed to submit decision.',
      );
      setBusy(false);
    }
  };
 
  return (
    <div className="oauth-consent">
      <div className="oauth-consent__card">
        <h1 className="oauth-consent__title">Authorize {clientName}</h1>
        <p className="oauth-consent__lede">
          <strong>{clientName}</strong> is requesting access to your Videfy
          account.
        </p>
 
        {missingFields ? (
          <div className="oauth-consent__error">
            This authorization request is missing required parameters.
            Please return to the application that opened this page and try
            again.
          </div>
        ) : (
          <>
            <div className="oauth-consent__section">
              <h2>It will be able to:</h2>
              <ul>
                {scopes.includes('videfy.read') && (
                  <li>Read your projects, scenes and library</li>
                )}
                {scopes.includes('videfy.write') && (
                  <>
                    <li>Create and edit projects on your behalf</li>
                    <li>
                      Generate images, videos and voiceovers
                      <span className="oauth-consent__hint">
                        {' '}
                        — uses your Videfy credits
                      </span>
                    </li>
                  </>
                )}
              </ul>
            </div>
 
            <div className="oauth-consent__section oauth-consent__meta">
              <div>
                <span className="oauth-consent__label">Redirects to</span>
                <code>{redirectUri}</code>
              </div>
            </div>
 
            {error && <div className="oauth-consent__error">{error}</div>}
 
            <div className="oauth-consent__actions">
              <button
                type="button"
                className="oauth-consent__btn oauth-consent__btn--secondary"
                disabled={busy}
                onClick={() => submit('deny')}
              >
                Deny
              </button>
              <button
                type="button"
                className="oauth-consent__btn oauth-consent__btn--primary"
                disabled={busy}
                onClick={() => submit('allow')}
              >
                {busy ? 'Authorizing…' : 'Authorize'}
              </button>
            </div>
 
            <p className="oauth-consent__footer">
              You can revoke access at any time from{' '}
              <a href="/settings/integrations">Settings · Integrations</a>.
            </p>
          </>
        )}
      </div>
    </div>
  );
};
 
export default OAuthConsent;

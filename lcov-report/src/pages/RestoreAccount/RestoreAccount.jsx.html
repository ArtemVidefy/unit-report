import React, { useState } from 'react';
import { useNavigate } from 'react-router-dom';
import { useDispatch } from 'react-redux';
import { NavLink } from 'react-router-dom';
import { useAuth } from '@clerk/clerk-react';
import { cancelDeleteAccount } from 'utils/users';
import { refreshUser, logout } from '../../redux/auth';
import videfyLogo from '../../images/videfyLogo.png';
import styles from './RestoreAccount.module.scss';
import { user as userSelector } from '../../redux/auth/selectors';
import { useSelector } from 'react-redux';
import { useError } from 'contexts/ErrorContext';
 
const getRemainingDays = (deletionRequestedAt) => {
  if (!deletionRequestedAt) return null;
  const requestedDate = new Date(deletionRequestedAt);
  const deletionDate = new Date(requestedDate.getTime() + 30 * 24 * 60 * 60 * 1000);
  const diffMs = deletionDate - Date.now();
  return Math.max(0, Math.floor(diffMs / (24 * 60 * 60 * 1000)));
};
 
const RestoreAccount = () => {
  const { showError, showInfoPositive } = useError();
  const navigate = useNavigate();
  const dispatch = useDispatch();
  const { signOut } = useAuth();
  const [isLoading, setIsLoading] = useState(false);
  const user = useSelector(userSelector);
  const remainingDays = getRemainingDays(user?.deletionRequestedAt);
  const [restoreAccountModalOpen, setRestoreAccountModalOpen] = useState(false);
 
  const handleLogout = async () => {
    await signOut();
    dispatch(logout());
  };
 
  const handleRestoreAccount = async () => {
    setIsLoading(true);
    try {
      await cancelDeleteAccount();
      showInfoPositive('Account restored successfully');
      await dispatch(refreshUser()).unwrap();
      navigate('/home');
    } catch (err) {
      console.error('handleRestoreAccount error:', err);
      showError('Failed to restore account');
    } finally {
      setIsLoading(false);
    }
  };
 
  return (
    <>
      <header className={styles.pageHeader}>
        <NavLink to="/">
          <img className={styles.header_logo} src={videfyLogo} alt="Videfy Logo" />
        </NavLink>
      </header>
      <section className={styles.wrap}>
        <div className={styles.content}>
          <h1 className={styles.title}>Account Deletion Scheduled</h1>
          <p className={styles.restoreAccountTextLine1}>
            Your account deletion request was submitted on {new Date(user?.deletionRequestedAt).toLocaleDateString('en-US', { month: 'long', day: 'numeric', year: 'numeric' })}
          </p>
          <p className={styles.restoreAccountTextLine2}>
            Your account will be deleted in <span>{remainingDays} {remainingDays === 1 ? 'day' : 'days'}</span>. Restore it anytime before then.
          </p>
          <div className={styles.actions}>
            <button
              type="button"
              className={styles.logoutButton}
              onClick={handleLogout}
            >
              Log out
            </button>
            <button
              type="button"
              className={styles.restoreButton}
              onClick={() => setRestoreAccountModalOpen(true)}
              disabled={isLoading}
            >
              {isLoading ? 'Restoring...' : 'Restore account'}
            </button>
          </div>
        </div>
      </section>
      {restoreAccountModalOpen && (
        <div
          className={styles.restoreAccountOverlay}
          onClick={() => setRestoreAccountModalOpen(false)}
        >
          <div
            className={styles.restoreAccountModal}
            onClick={(e) => e.stopPropagation()}
            role="dialog"
            aria-modal="true"
            aria-labelledby="restore-account-title"
          >
            <div className={styles.restoreAccountModalTop}>
              <h3 id="restore-account-title" className={styles.restoreAccountTitle}>
                Restore your account
              </h3>
              <p className={styles.restoreAccountText}>
                Are you sure you want to restore your account?
              </p>
            </div>
            <div className={styles.restoreAccountActions}>
              <button
                type="button"
                className={styles.restoreAccountButtonSecondary}
                onClick={() => setRestoreAccountModalOpen(false)}
              >
                No
              </button>
              <button
                type="button"
                className={styles.restoreAccountButtonPrimary}
                onClick={() => {
                  handleRestoreAccount();
                  setRestoreAccountModalOpen(false);
                }}
              >
                Yes
              </button>
            </div>
          </div>
        </div>
      )}
    </>
  );
};
 
export { RestoreAccount };

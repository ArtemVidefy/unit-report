import { useNavigate } from 'react-router-dom';
import { useState, useEffect, useMemo } from 'react';
import styles from './WelcomePage.module.scss';
import { WelcomeRingIcon } from 'components/Icons';
import videfyLogo from '../../images/videfyLogo.png';
import welcomeTexts from 'data/welcomeTexts.json';
 
const WelcomePage = () => {
  const navigate = useNavigate();
 
  // Generate random initial index once
  const initialRandomIndex = useMemo(
    () => Math.floor(Math.random() * welcomeTexts.textVariations.length),
    []
  );
 
  // State for current text variation - start with random index
  const [currentTextIndex, setCurrentTextIndex] = useState(initialRandomIndex);
  const [currentText, setCurrentText] = useState(
    welcomeTexts.textVariations[initialRandomIndex]
  );
 
  // Split text into lines from current variation
  const line1 = currentText.lines[0];
  const line2 = currentText.lines[1];
  const line3 = currentText.lines[2];
  const line4 = currentText.lines[3];
  const highlightedText = currentText.highlightedText;
 
  const [displayLine1, setDisplayLine1] = useState('');
  const [displayLine2, setDisplayLine2] = useState('');
  const [displayLine3, setDisplayLine3] = useState('');
  const [displayLine4, setDisplayLine4] = useState('');
  const [line1Done, setLine1Done] = useState(false);
  const [line2Done, setLine2Done] = useState(false);
  const [line3Done, setLine3Done] = useState(false);
  const [line4Done, setLine4Done] = useState(false);
  const [currentLine1Index, setCurrentLine1Index] = useState(0);
  const [currentLine2Index, setCurrentLine2Index] = useState(0);
  const [currentLine3Index, setCurrentLine3Index] = useState(0);
  const [currentLine4Index, setCurrentLine4Index] = useState(0);
  const [highlightedIndex, setHighlightedIndex] = useState(0);
  const [isTypingHighlighted, setIsTypingHighlighted] = useState(false);
  const [allTypingDone, setAllTypingDone] = useState(false);
  const [shouldReset, setShouldReset] = useState(false);
 
  // Function to get next text variation
  const getNextTextVariation = () => {
    if (welcomeTexts.settings.rotationMode === 'random') {
      let nextIndex;
      do {
        nextIndex = Math.floor(
          Math.random() * welcomeTexts.textVariations.length
        );
      } while (
        nextIndex === currentTextIndex &&
        welcomeTexts.textVariations.length > 1
      );
      return nextIndex;
    } else {
      // Sequential mode
      return (currentTextIndex + 1) % welcomeTexts.textVariations.length;
    }
  };
 
  // Reset all states to start the animation again
  const resetAnimation = () => {
    // Select next text variation
    const nextIndex = getNextTextVariation();
    setCurrentTextIndex(nextIndex);
    setCurrentText(welcomeTexts.textVariations[nextIndex]);
 
    // Reset animation states
    setDisplayLine1('');
    setDisplayLine2('');
    setDisplayLine3('');
    setDisplayLine4('');
    setLine1Done(false);
    setLine2Done(false);
    setLine3Done(false);
    setLine4Done(false);
    setCurrentLine1Index(0);
    setCurrentLine2Index(0);
    setCurrentLine3Index(0);
    setCurrentLine4Index(0);
    setHighlightedIndex(0);
    setIsTypingHighlighted(false);
    setAllTypingDone(false);
    setShouldReset(false);
  };
 
  // Type line 1
  useEffect(() => {
    if (currentLine1Index < line1.length) {
      const typingTimer = setTimeout(() => {
        setDisplayLine1(prevText => prevText + line1[currentLine1Index]);
        setCurrentLine1Index(prevIndex => prevIndex + 1);
      }, welcomeTexts.settings.typingSpeed);
 
      return () => clearTimeout(typingTimer);
    } else {
      setLine1Done(true);
    }
  }, [currentLine1Index, line1]);
 
  // Type line 2 after line 1 is done
  useEffect(() => {
    if (line1Done && currentLine2Index < line2.length) {
      const typingTimer = setTimeout(() => {
        setDisplayLine2(prevText => prevText + line2[currentLine2Index]);
        setCurrentLine2Index(prevIndex => prevIndex + 1);
      }, welcomeTexts.settings.typingSpeed);
 
      return () => clearTimeout(typingTimer);
    } else if (line1Done && currentLine2Index === line2.length) {
      setLine2Done(true);
    }
  }, [line1Done, currentLine2Index, line2]);
 
  // Type line 3 after line 2 is done
  useEffect(() => {
    if (line2Done && currentLine3Index < line3.length) {
      const typingTimer = setTimeout(() => {
        setDisplayLine3(prevText => prevText + line3[currentLine3Index]);
        setCurrentLine3Index(prevIndex => prevIndex + 1);
      }, welcomeTexts.settings.typingSpeed);
 
      return () => clearTimeout(typingTimer);
    } else if (line2Done && currentLine3Index === line3.length) {
      setLine3Done(true);
    }
  }, [line2Done, currentLine3Index, line3]);
 
  // Type line 4 after line 3 is done
  useEffect(() => {
    if (line3Done && currentLine4Index < line4.length) {
      const typingTimer = setTimeout(() => {
        setDisplayLine4(prevText => prevText + line4[currentLine4Index]);
        setCurrentLine4Index(prevIndex => prevIndex + 1);
      }, welcomeTexts.settings.typingSpeed);
 
      return () => clearTimeout(typingTimer);
    } else if (line3Done && currentLine4Index === line4.length) {
      setLine4Done(true);
    }
  }, [line3Done, currentLine4Index, line4]);
 
  // Type highlighted text after line 4 is done
  useEffect(() => {
    if (line4Done && !isTypingHighlighted) {
      setIsTypingHighlighted(true);
    } else if (
      isTypingHighlighted &&
      highlightedIndex < highlightedText.length
    ) {
      const typingTimer = setTimeout(() => {
        setHighlightedIndex(prevIndex => prevIndex + 1);
      }, welcomeTexts.settings.typingSpeed);
 
      return () => clearTimeout(typingTimer);
    } else if (
      isTypingHighlighted &&
      highlightedIndex === highlightedText.length
    ) {
      // Set allTypingDone to true when all text is typed
      const finalTimer = setTimeout(() => {
        setAllTypingDone(true);
      }, welcomeTexts.settings.finalDelay); // Small delay after typing finishes
 
      return () => clearTimeout(finalTimer);
    }
  }, [line4Done, isTypingHighlighted, highlightedIndex, highlightedText]);
 
  // Set up the reset cycle after all typing is done
  useEffect(() => {
    if (allTypingDone && !shouldReset) {
      const resetTimer = setTimeout(() => {
        setShouldReset(true);
      }, welcomeTexts.settings.resetDelay); // Delay before resetting
 
      return () => clearTimeout(resetTimer);
    }
 
    if (shouldReset) {
      resetAnimation();
    }
  }, [allTypingDone, shouldReset]);
 
  const handleLogin = () => {
    navigate('/login');
  };
 
  const handleSignUp = () => {
    navigate('/signUp');
  };
 
  // Determine where to show the cursor
  const showCursor = () => {
    if (!line1Done) {
      return 1;
    } else if (!line2Done) {
      return 2;
    } else if (!line3Done) {
      return 3;
    } else if (!line4Done) {
      return 4;
    } else if (highlightedIndex < highlightedText.length) {
      return 5;
    }
    return 0; // Don't show cursor when all typing is done
  };
 
  return (
    <section className={styles.wrapper}>
      <header className={styles.header}>
        <img className={styles.header_logo} src={videfyLogo} alt="Videfy Logo" onClick={() => window.location.href = 'https://videfy.ai'} />
        <h1 className={styles.header_text}>Videfy.ai</h1>
      </header>
      <main className={styles.main}>
        <div className={styles.heroContent}>
          <div className={styles.heroText}>
            <div className={styles.typingLine}>
              {displayLine1}
              {showCursor() === 1 && (
                <span className={styles.cursorBlink}></span>
              )}
            </div>
            <div className={styles.typingLine}>
              {displayLine2}
              {showCursor() === 2 && (
                <span className={styles.cursorBlink}></span>
              )}
            </div>
            <div className={styles.typingLine}>
              {displayLine3}
              {showCursor() === 3 && (
                <span className={styles.cursorBlink}></span>
              )}
            </div>
            <div className={styles.typingLine}>
              {displayLine4}
              {line4Done && line4.length > 0 && (
                <span className={styles.spaceCharacter}>&nbsp;</span>
              )}
              {isTypingHighlighted && (
                <span className={styles.highlightedText}>
                  {highlightedText.substring(0, highlightedIndex)}
                </span>
              )}
              {showCursor() > 3 && <span className={styles.cursorBlink}></span>}
            </div>
 
            {/* Only show the icon when all typing is done */}
            {/* {allTypingDone && (
              <div className={styles.welcomeRingIcon}>
                <WelcomeRingIcon />
              </div>
            )} */}
          </div>
        </div>
        <div className={styles.authContainer}>
          <div className={styles.authContent}>
            <h2 className={styles.title}>Let&apos;s start</h2>
            <div className={styles.authButtons}>
              <button className={styles.authButton} onClick={handleLogin}>
                Log in
              </button>
              <button className={styles.authButton} onClick={handleSignUp}>
                Sign up
              </button>
            </div>
          </div>
          <div className={styles.usageLinks}>
            <a
              className={styles.usageLink}
              href="https://docs.google.com/document/d/1D-xBm7SRuzAEsV3HBwsnNFNgdtecpUCrHJGsrfvn300/edit?tab=t.0"
              target="_blank"
              rel="noopener noreferrer"
            >
              Terms of Use
            </a>
            <div className={styles.separator}>•</div>
            <a
              className={styles.usageLink}
              href="https://docs.google.com/document/d/1yEw-Np1nf6WfSdhHPcZRCvvKAfijNqRu3ENMZBjH7L8/edit?tab=t.0"
              target="_blank"
              rel="noopener noreferrer"
            >
              Cookie Policy
            </a>
            <span className={styles.separator}>•</span>
            <a
              className={styles.usageLink}
              href="https://docs.google.com/document/d/1SgXznj-3g76Bb4WG6pRcTcd1XDfQ-WQfVlw-iPhiHlc/edit?tab=t.0"
              target="_blank"
              rel="noopener noreferrer"
            >
              Privacy Policy
            </a>
          </div>
        </div>
      </main>
      <div className="stain"></div>
      <div className="stainBottom"></div>
    </section>
  );
};
 
export { WelcomePage };
